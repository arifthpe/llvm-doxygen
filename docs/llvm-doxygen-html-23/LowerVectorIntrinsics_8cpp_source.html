17bool llvm::lowerUnaryVectorIntrinsicAsLoop(Module &M, CallInst *CI) {
18 Type *ArgTy = CI->getArgOperand(0)->getType();
19 VectorType *VecTy = cast<VectorType>(ArgTy);
20
21 BasicBlock *PreLoopBB = CI->getParent();
22 BasicBlock *PostLoopBB = nullptr;
23 Function *ParentFunc = PreLoopBB->getParent();
24 LLVMContext &Ctx = PreLoopBB->getContext();
25 Type *IdxTy = M.getDataLayout().getIndexType(Ctx, 0);
26
27 PostLoopBB = PreLoopBB->splitBasicBlock(CI);
28 BasicBlock *LoopBB = BasicBlock::Create(Ctx, "", ParentFunc, PostLoopBB);
29 PreLoopBB->getTerminator()->setSuccessor(0, LoopBB);
30
31 // Loop preheader
32 IRBuilder<> PreLoopBuilder(PreLoopBB->getTerminator());
33 Value *LoopEnd =
34 PreLoopBuilder.CreateElementCount(IdxTy, VecTy->getElementCount());
35
36 // Loop body
37 IRBuilder<> LoopBuilder(LoopBB);
38
39 PHINode *LoopIndex = LoopBuilder.CreatePHI(IdxTy, 2);
40 LoopIndex->addIncoming(ConstantInt::get(IdxTy, 0U), PreLoopBB);
41 PHINode *Vec = LoopBuilder.CreatePHI(VecTy, 2);
42 Vec->addIncoming(CI->getArgOperand(0), PreLoopBB);
43
44 Value *Elem = LoopBuilder.CreateExtractElement(Vec, LoopIndex);
45 Function *Exp = Intrinsic::getOrInsertDeclaration(&M, CI->getIntrinsicID(),
46 VecTy->getElementType());
47 Value *Res = LoopBuilder.CreateCall(Exp, Elem);
48 Value *NewVec = LoopBuilder.CreateInsertElement(Vec, Res, LoopIndex);
49 Vec->addIncoming(NewVec, LoopBB);
50
51 Value *One = ConstantInt::get(IdxTy, 1U);
52 Value *NextLoopIndex = LoopBuilder.CreateAdd(LoopIndex, One);
53 LoopIndex->addIncoming(NextLoopIndex, LoopBB);
54
55 Value *ExitCond =
56 LoopBuilder.CreateICmp(CmpInst::ICMP_EQ, NextLoopIndex, LoopEnd);
57 LoopBuilder.CreateCondBr(ExitCond, PostLoopBB, LoopBB);
58
59 CI->replaceAllUsesWith(NewVec);
60 CI->eraseFromParent();
61 return true;
62}
