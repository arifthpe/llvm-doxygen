75/// Try to fold a select-based split cttz pattern into a single full-width cttz.
76///
77/// %lo = trunc iN %val to i(N/2)
78/// %cmp = icmp eq i(N/2) %lo, 0
79/// %shr = lshr iN %val, N/2
80/// %hi = trunc iN %shr to i(N/2)
81/// %cttz_hi = call i(N/2) @llvm.cttz.i(N/2)(i(N/2) %hi, ...)
82/// %hi_plus = add/or_disjoint i(N/2) %cttz_hi, N/2
83/// %cttz_lo = call i(N/2) @llvm.cttz.i(N/2)(i(N/2) %lo, ...)
84/// %result = select i1 %cmp, i(N/2) %hi_plus, i(N/2) %cttz_lo
85/// -->
86/// %cttz_wide = call iN @llvm.cttz.iN(iN %val, i1 false)
87/// %result = trunc iN %cttz_wide to i(N/2)
88/// Alive proof (for i64/i32): https://alive2.llvm.org/ce/z/-s14-s
89// TrueVal/FalseVal are pre-normalized by the caller to the EQ/NE cases.
90static bool foldSelectSplitCTTZ(Instruction &I, Value *LoTrunc, Value *HiResult,
91 Value *LoResult, Type *HalfTy) {
92 unsigned HalfWidth = HalfTy->getIntegerBitWidth();
93 unsigned FullWidth = HalfWidth * 2;
94
95 // LoTrunc: trunc iN SrcVal to i(N/2)
96 Value *SrcVal;
97 if (!match(LoTrunc, m_Trunc(m_Value(SrcVal))))
98 return false;
99 if (!SrcVal->getType()->isIntegerTy(FullWidth))
100 return false;
101
102 // LoResult: cttz(trunc(SrcVal), _), must use same truncated value
103 if (!match(LoResult, m_OneUse(m_Cttz(m_Specific(LoTrunc), m_Value()))))
104 return false;
105
106 // HiResult: add/or_disjoint(cttz(trunc(lshr(SrcVal, N/2)), _), N/2)
107 Value *CttzHiCall;
108 if (!match(HiResult, m_OneUse(m_AddLike(m_Value(CttzHiCall),
109 m_SpecificInt(HalfWidth)))))
110 return false;
111
112 Value *HiCttzArg;
113 if (!match(CttzHiCall, m_OneUse(m_Cttz(m_Value(HiCttzArg), m_Value()))))
114 return false;
115
116 if (!match(HiCttzArg,
117 m_Trunc(m_LShr(m_Specific(SrcVal), m_SpecificInt(HalfWidth)))))
118 return false;
119
120 // Match successful.
121 IRBuilder<> Builder(&I);
122 Value *CttzWide = Builder.CreateIntrinsic(
123 Intrinsic::cttz, {SrcVal->getType()}, {SrcVal, Builder.getFalse()});
124 Value *Trunc = Builder.CreateTrunc(CttzWide, HalfTy);
125
126 I.replaceAllUsesWith(Trunc);
127 ++NumSelectCTTZFolded;
128 return true;
129}

131/// Same as foldSelectSplitCTTZ but for leading zeros (ctlz).
132///
133/// %shr = lshr iN %val, N/2
134/// %hi = trunc iN %shr to i(N/2)
135/// %cmp = icmp eq i(N/2) %hi, 0 (or icmp eq iN %shr, 0)
136/// %lo = trunc iN %val to i(N/2)
137/// %ctlz_lo = call i(N/2) @llvm.ctlz.i(N/2)(i(N/2) %lo, ...)
138/// %lo_plus = add/or_disjoint i(N/2) %ctlz_lo, N/2
139/// %ctlz_hi = call i(N/2) @llvm.ctlz.i(N/2)(i(N/2) %hi, ...)
140/// %result = select i1 %cmp, i(N/2) %lo_plus, i(N/2) %ctlz_hi
141/// -->
142/// %ctlz_wide = call iN @llvm.ctlz.iN(iN %val, i1 false)
143/// %result = trunc iN %ctlz_wide to i(N/2)
144///
145/// Alive proof (for i64/i32): https://alive2.llvm.org/ce/z/WfQepH
146// TrueVal/FalseVal are pre-normalized by the caller to the EQ/NE cases.
147static bool foldSelectSplitCTLZ(Instruction &I, Value *HiPart, Value *LoResult,
148 Value *HiResult, Type *HalfTy) {
149 unsigned HalfWidth = HalfTy->getIntegerBitWidth();
150 unsigned FullWidth = HalfWidth * 2;
151
152 // Extract SrcVal from HiPart: either trunc(lshr(SrcVal, N/2)) or
153 // lshr(SrcVal, N/2)
154 Value *SrcVal;
155 if (match(HiPart, m_Trunc(m_Value(SrcVal))))
156 HiPart = SrcVal;
157
158 if (!match(HiPart, m_LShr(m_Value(SrcVal), m_SpecificInt(HalfWidth))))
159 return false;
160 if (!SrcVal->getType()->isIntegerTy(FullWidth))
161 return false;
162
163 // HiResult: ctlz(trunc(lshr(SrcVal, N/2)), _)
164 Value *HiCtlzArg;
165 if (!match(HiResult, m_OneUse(m_Ctlz(m_Value(HiCtlzArg), m_Value()))))
166 return false;
167
168 if (!match(HiCtlzArg,
169 m_Trunc(m_LShr(m_Specific(SrcVal), m_SpecificInt(HalfWidth)))))
170 return false;
171
172 // LoResult: add/or_disjoint(ctlz(trunc(SrcVal), _), N/2)
173 Value *CtlzLoCall;
174 if (!match(LoResult, m_OneUse(m_AddLike(m_Value(CtlzLoCall),
175 m_SpecificInt(HalfWidth)))))
176 return false;
177
178 Value *LoCtlzArg;
179 if (!match(CtlzLoCall, m_OneUse(m_Ctlz(m_Value(LoCtlzArg), m_Value()))))
180 return false;
181
182 if (!match(LoCtlzArg, m_Trunc(m_Specific(SrcVal))))
183 return false;
184
185 // Match successful.
186 IRBuilder<> Builder(&I);
187 Value *CtlzWide = Builder.CreateIntrinsic(
188 Intrinsic::ctlz, {SrcVal->getType()}, {SrcVal, Builder.getFalse()});
189 Value *Trunc = Builder.CreateTrunc(CtlzWide, HalfTy);
190
191 I.replaceAllUsesWith(Trunc);
192 ++NumSelectCTLZFolded;
193 return true;
194}

196/// Common entry point for folding select-based split cttz/ctlz patterns.
197/// Performs the initial select and type matching shared by both transforms,
198/// then delegates to foldSelectSplitCTTZ and foldSelectSplitCTLZ.
199static bool foldSelectSplitCTLZCTTZ(Instruction &I) {
200 Value *Cond, *TrueVal, *FalseVal;
201 if (!match(&I, m_Select(m_Value(Cond), m_Value(TrueVal), m_Value(FalseVal))))
202 return false;
203
204 Type *Ty = I.getType();
205 if (!Ty->isIntegerTy())
206 return false;
207
208 // Bail out on very small types (i1, i2): the full-width cttz/ctlz can return
209 // values not representable in the half type (e.g., cttz.i4 can return 4,
210 // which doesn't fit in i2).
211 if (Ty->getIntegerBitWidth() <= 2)
212 return false;
213
214 CmpPredicate Pred;
215 Value *CmpOp;
216 if (!match(Cond, m_ICmp(Pred, m_Value(CmpOp), m_ZeroInt())) ||
217 !ICmpInst::isEquality(Pred))
218 return false;
219
220 // Canonicalize select operands.
221 if (Pred == CmpInst::ICMP_NE)
222 std::swap(TrueVal, FalseVal);
223
224 return foldSelectSplitCTTZ(I, CmpOp, TrueVal, FalseVal, Ty) ||
225 foldSelectSplitCTLZ(I, CmpOp, TrueVal, FalseVal, Ty);
226}

228/// Match a pattern for a bitwise funnel/rotate operation that partially guards
229/// against undefined behavior by branching around the funnel-shift/rotation
230/// when the shift amount is 0.
231static bool foldGuardedFunnelShift(Instruction &I, const DominatorTree &DT) {
232 if (I.getOpcode() != Instruction::PHI || I.getNumOperands() != 2)
233 return false;
234
235 // As with the one-use checks below, this is not strictly necessary, but we
236 // are being cautious to avoid potential perf regressions on targets that
237 // do not actually have a funnel/rotate instruction (where the funnel shift
238 // would be expanded back into math/shift/logic ops).
239 if (!isPowerOf2_32(I.getType()->getScalarSizeInBits()))
240 return false;
241
242 // Match V to funnel shift left/right and capture the source operands and
243 // shift amount.
244 auto matchFunnelShift = [](Value *V, Value *&ShVal0, Value *&ShVal1,
245 Value *&ShAmt) {
246 unsigned Width = V->getType()->getScalarSizeInBits();
247
248 // fshl(ShVal0, ShVal1, ShAmt)
249 // == (ShVal0 << ShAmt) | (ShVal1 >> (Width -ShAmt))
250 if (match(V, m_OneUse(m_c_Or(
251 m_Shl(m_Value(ShVal0), m_Value(ShAmt)),
252 m_LShr(m_Value(ShVal1), m_Sub(m_SpecificInt(Width),
253 m_Deferred(ShAmt))))))) {
254 return Intrinsic::fshl;
255 }
256
257 // fshr(ShVal0, ShVal1, ShAmt)
258 // == (ShVal0 >> ShAmt) | (ShVal1 << (Width - ShAmt))
259 if (match(V,
260 m_OneUse(m_c_Or(m_Shl(m_Value(ShVal0), m_Sub(m_SpecificInt(Width),
261 m_Value(ShAmt))),
262 m_LShr(m_Value(ShVal1), m_Deferred(ShAmt)))))) {
263 return Intrinsic::fshr;
264 }
265
266 return Intrinsic::not_intrinsic;
267 };
268
269 // One phi operand must be a funnel/rotate operation, and the other phi
270 // operand must be the source value of that funnel/rotate operation:
271 // phi [ rotate(RotSrc, ShAmt), FunnelBB ], [ RotSrc, GuardBB ]
272 // phi [ fshl(ShVal0, ShVal1, ShAmt), FunnelBB ], [ ShVal0, GuardBB ]
273 // phi [ fshr(ShVal0, ShVal1, ShAmt), FunnelBB ], [ ShVal1, GuardBB ]
274 PHINode &Phi = cast<PHINode>(I);
275 unsigned FunnelOp = 0, GuardOp = 1;
276 Value *P0 = Phi.getOperand(0), *P1 = Phi.getOperand(1);
277 Value *ShVal0, *ShVal1, *ShAmt;
278 Intrinsic::ID IID = matchFunnelShift(P0, ShVal0, ShVal1, ShAmt);
279 if (IID == Intrinsic::not_intrinsic ||
280 (IID == Intrinsic::fshl && ShVal0 != P1) ||
281 (IID == Intrinsic::fshr && ShVal1 != P1)) {
282 IID = matchFunnelShift(P1, ShVal0, ShVal1, ShAmt);
283 if (IID == Intrinsic::not_intrinsic ||
284 (IID == Intrinsic::fshl && ShVal0 != P0) ||
285 (IID == Intrinsic::fshr && ShVal1 != P0))
286 return false;
287 assert((IID == Intrinsic::fshl || IID == Intrinsic::fshr) &&
288 "Pattern must match funnel shift left or right");
289 std::swap(FunnelOp, GuardOp);
290 }
291
292 // The incoming block with our source operand must be the "guard" block.
293 // That must contain a cmp+branch to avoid the funnel/rotate when the shift
294 // amount is equal to 0. The other incoming block is the block with the
295 // funnel/rotate.
296 BasicBlock *GuardBB = Phi.getIncomingBlock(GuardOp);
297 BasicBlock *FunnelBB = Phi.getIncomingBlock(FunnelOp);
298 Instruction *TermI = GuardBB->getTerminator();
299
300 // Ensure that the shift values dominate each block.
301 if (!DT.dominates(ShVal0, TermI) || !DT.dominates(ShVal1, TermI))
302 return false;
303
304 BasicBlock *PhiBB = Phi.getParent();
305 if (!match(TermI, m_Br(m_SpecificICmp(CmpInst::ICMP_EQ, m_Specific(ShAmt),
306 m_ZeroInt()),
307 m_SpecificBB(PhiBB), m_SpecificBB(FunnelBB))))
308 return false;
309
310 IRBuilder<> Builder(PhiBB, PhiBB->getFirstInsertionPt());
311
312 if (ShVal0 == ShVal1)
313 ++NumGuardedRotates;
314 else
315 ++NumGuardedFunnelShifts;
316
317 // If this is not a rotate then the select was blocking poison from the
318 // 'shift-by-zero' non-TVal, but a funnel shift won't - so freeze it.
319 bool IsFshl = IID == Intrinsic::fshl;
320 if (ShVal0 != ShVal1) {
321 if (IsFshl && !llvm::isGuaranteedNotToBePoison(ShVal1))
322 ShVal1 = Builder.CreateFreeze(ShVal1);
323 else if (!IsFshl && !llvm::isGuaranteedNotToBePoison(ShVal0))
324 ShVal0 = Builder.CreateFreeze(ShVal0);
325 }
326
327 // We matched a variation of this IR pattern:
328 // GuardBB:
329 // %cmp = icmp eq i32 %ShAmt, 0
330 // br i1 %cmp, label %PhiBB, label %FunnelBB
331 // FunnelBB:
332 // %sub = sub i32 32, %ShAmt
333 // %shr = lshr i32 %ShVal1, %sub
334 // %shl = shl i32 %ShVal0, %ShAmt
335 // %fsh = or i32 %shr, %shl
336 // br label %PhiBB
337 // PhiBB:
338 // %cond = phi i32 [ %fsh, %FunnelBB ], [ %ShVal0, %GuardBB ]
339 // -->
340 // llvm.fshl.i32(i32 %ShVal0, i32 %ShVal1, i32 %ShAmt)
341 Phi.replaceAllUsesWith(
342 Builder.CreateIntrinsic(IID, Phi.getType(), {ShVal0, ShVal1, ShAmt}));
343 return true;
344}

346/// This is used by foldAnyOrAllBitsSet() to capture a source value (Root) and
347/// the bit indexes (Mask) needed by a masked compare. If we're matching a chain
348/// of 'and' ops, then we also need to capture the fact that we saw an
349/// "and X, 1", so that's an extra return value for that case.
350namespace {
351struct MaskOps {
352 Value *Root = nullptr;
353 APInt Mask;
354 bool MatchAndChain;
355 bool FoundAnd1 = false;
356
357 MaskOps(unsigned BitWidth, bool MatchAnds)
358 : Mask(APInt::getZero(BitWidth)), MatchAndChain(MatchAnds) {}
359};
360} // namespace

362/// This is a recursive helper for foldAnyOrAllBitsSet() that walks through a
363/// chain of 'and' or 'or' instructions looking for shift ops of a common source
364/// value. Examples:
365/// or (or (or X, (X >> 3)), (X >> 5)), (X >> 8)
366/// returns { X, 0x129 }
367/// and (and (X >> 1), 1), (X >> 4)
368/// returns { X, 0x12 }
369static bool matchAndOrChain(Value *V, MaskOps &MOps) {
370 Value *Op0, *Op1;
371 if (MOps.MatchAndChain) {
372 // Recurse through a chain of 'and' operands. This requires an extra check
373 // vs. the 'or' matcher: we must find an "and X, 1" instruction somewhere
374 // in the chain to know that all of the high bits are cleared.
375 if (match(V, m_And(m_Value(Op0), m_One()))) {
376 MOps.FoundAnd1 = true;
377 return matchAndOrChain(Op0, MOps);
378 }
379 if (match(V, m_And(m_Value(Op0), m_Value(Op1))))
380 return matchAndOrChain(Op0, MOps) && matchAndOrChain(Op1, MOps);
381 } else {
382 // Recurse through a chain of 'or' operands.
383 if (match(V, m_Or(m_Value(Op0), m_Value(Op1))))
384 return matchAndOrChain(Op0, MOps) && matchAndOrChain(Op1, MOps);
385 }
386
387 // We need a shift-right or a bare value representing a compare of bit 0 of
388 // the original source operand.
389 Value *Candidate;
390 const APInt *BitIndex = nullptr;
391 if (!match(V, m_LShr(m_Value(Candidate), m_APInt(BitIndex))))
392 Candidate = V;
393
394 // Initialize result source operand.
395 if (!MOps.Root)
396 MOps.Root = Candidate;
397
398 // The shift constant is out-of-range? This code hasn't been simplified.
399 if (BitIndex && BitIndex->uge(MOps.Mask.getBitWidth()))
400 return false;
401
402 // Fill in the mask bit derived from the shift constant.
403 MOps.Mask.setBit(BitIndex ? BitIndex->getZExtValue() : 0);
404 return MOps.Root == Candidate;
405}

407/// Match patterns that correspond to "any-bits-set" and "all-bits-set".
408/// These will include a chain of 'or' or 'and'-shifted bits from a
409/// common source value:
410/// and (or (lshr X, C), ...), 1 --> (X & CMask) != 0
411/// and (and (lshr X, C), ...), 1 --> (X & CMask) == CMask
412/// Note: "any-bits-clear" and "all-bits-clear" are variations of these patterns
413/// that differ only with a final 'not' of the result. We expect that final
414/// 'not' to be folded with the compare that we create here (invert predicate).
415static bool foldAnyOrAllBitsSet(Instruction &I) {
416 // The 'any-bits-set' ('or' chain) pattern is simpler to match because the
417 // final "and X, 1" instruction must be the final op in the sequence.
418 bool MatchAllBitsSet;
419 bool MatchTrunc;
420 Value *X;
421 if (I.getType()->isIntOrIntVectorTy(1)) {
422 if (match(&I, m_Trunc(m_OneUse(m_And(m_Value(), m_Value())))))
423 MatchAllBitsSet = true;
424 else if (match(&I, m_Trunc(m_OneUse(m_Or(m_Value(), m_Value())))))
425 MatchAllBitsSet = false;
426 else
427 return false;
428 MatchTrunc = true;
429 X = I.getOperand(0);
430 } else {
431 if (match(&I, m_c_And(m_OneUse(m_And(m_Value(), m_Value())), m_Value()))) {
432 X = &I;
433 MatchAllBitsSet = true;
434 } else if (match(&I,
435 m_And(m_OneUse(m_Or(m_Value(), m_Value())), m_One()))) {
436 X = I.getOperand(0);
437 MatchAllBitsSet = false;
438 } else
439 return false;
440 MatchTrunc = false;
441 }
442 Type *Ty = X->getType();
443
444 MaskOps MOps(Ty->getScalarSizeInBits(), MatchAllBitsSet);
445 if (!matchAndOrChain(X, MOps) ||
446 (MatchAllBitsSet && !MatchTrunc && !MOps.FoundAnd1))
447 return false;
448
449 // The pattern was found. Create a masked compare that replaces all of the
450 // shift and logic ops.
451 IRBuilder<> Builder(&I);
452 Constant *Mask = ConstantInt::get(Ty, MOps.Mask);
453 Value *And = Builder.CreateAnd(MOps.Root, Mask);
454 Value *Cmp = MatchAllBitsSet ? Builder.CreateICmpEQ(And, Mask)
455 : Builder.CreateIsNotNull(And);
456 Value *Zext = MatchTrunc ? Cmp : Builder.CreateZExt(Cmp, Ty);
457 I.replaceAllUsesWith(Zext);
458 ++NumAnyOrAllBitsSet;
459 return true;
460}

462/// Helper function to replace an instruction with a popcount intrinsic.
463/// This creates the ctpop intrinsic with an optional truncation appended at the
464/// end, and replaces all uses of the instruction.
465static void replaceWithPopCount(Instruction &I, Value *Root) {
466 LLVM_DEBUG(dbgs() << "Recognized popcount intrinsic\n");
467 Type *RootTy = Root->getType();
468 Type *OrigTy = I.getType();
469
470 IRBuilder<> Builder(&I);
471 Value *NewVal = Builder.CreateIntrinsic(Intrinsic::ctpop, RootTy, {Root});
472 if (OrigTy != RootTy) {
473 assert(RootTy->getScalarSizeInBits() > OrigTy->getScalarSizeInBits() &&
474 "Only truncation is supported for now");
475 NewVal = Builder.CreateTrunc(NewVal, OrigTy);
476 }
477 I.replaceAllUsesWith(NewVal);
478 ++NumPopCountRecognized;
479}

481// Matches the common innermost steps of the Hacker's Delight popcount idiom:
482// V = ((x + (x >> 4)) & 0x0F...)
483// x = (y & 0x33...) + ((y >> 2) & 0x33...) [or y - 3*((y>>2)&0x33...)]
484// y = Root - ((Root >> 1) & 0x55...)
485// This computes the popcount for each byte.
486// Returns Root on success, nullptr on failure.
487static Value *matchPopCountBytes(Value *V, unsigned Len, const DataLayout &DL) {
488 APInt Mask55 = APInt::getSplat(Len, APInt(8, 0x55));
489 APInt Mask33 = APInt::getSplat(Len, APInt(8, 0x33));
490 APInt Mask0F = APInt::getSplat(Len, APInt(8, 0x0F));
491
492 Value *Add2;
493 // Matching "((x + (x >> 4)) & 0x0F...)".
494 if (!match(V, m_And(m_c_Add(m_LShr(m_Value(Add2), m_SpecificInt(4)),
495 m_Deferred(Add2)),
496 m_SpecificInt(Mask0F))))
497 return nullptr;
498
499 Value *Sub1;
500 APInt NegThree(Len, -3, /*isSigned=*/true);
501 // Match
502 // x = (x & 0x33333333) + ((x >> 2) & 0x33333333)"
503 // Or
504 // x = x - 3*((x >> 2) & 0x33333333)
505 if (!match(Add2, m_c_Add(m_And(m_LShr(m_Value(Sub1), m_SpecificInt(2)),
506 m_SpecificInt(Mask33)),
507 m_And(m_Deferred(Sub1), m_SpecificInt(Mask33)))) &&
508 !match(Add2, m_Add(m_Mul(m_And(m_LShr(m_Value(Sub1), m_SpecificInt(2)),
509 m_SpecificInt(Mask33)),
510 m_SpecificInt(NegThree)),
511 m_Deferred(Sub1))))
512 return nullptr;
513
514 Value *Root, *LShr;
515 const APInt *AndMask;
516 // Matching "x - ((x >> 1) & 0x55...)".
517 if (!match(Sub1,
518 m_Sub(m_Value(Root), m_And(m_Value(LShr, m_LShr(m_Deferred(Root),
519 m_SpecificInt(1))),
520 m_APInt(AndMask)))))
521 return nullptr;
522
523 if (*AndMask != Mask55) {
524 // Accept a narrowed mask if missing bits are known zero in Root>>1.
525 if (!AndMask->isSubsetOf(Mask55))
526 return nullptr;
527 APInt NeededMask = Mask55 & ~*AndMask;
528 if (!MaskedValueIsZero(LShr, NeededMask, SimplifyQuery(DL)))
529 return nullptr;
530 }
531
532 return Root;
533}

535// Try to recognize below function as popcount intrinsic.
536// This is the "best" algorithm from
537// http://graphics.stanford.edu/~seander/bithacks.html#CountBitsSetParallel
538// Also used in TargetLowering::expandCTPOP().
539//
540// int popcount(unsigned int i) {
541// i = i - ((i >> 1) & 0x55555555);
542// i = (i & 0x33333333) + ((i >> 2) & 0x33333333);
543// i = ((i + (i >> 4)) & 0x0F0F0F0F);
544// return (i * 0x01010101) >> 24;
545// }
546static bool tryToRecognizePopCount(Instruction &I) {

548 return false;
549
550 Type *Ty = I.getType();
551 if (!Ty->isIntOrIntVectorTy())
552 return false;
553
554 unsigned Len = Ty->getScalarSizeInBits();
555 // Len==8 is handled by tryToRecognizePopCount2n3.
556 // FIXME: other irregular type lengths.
557 if (Len > 128 || Len <= 8 || Len % 8 != 0)
558 return false;
559
560 APInt Mask01 = APInt::getSplat(Len, APInt(8, 0x01));
561
562 Value *Op0 = I.getOperand(0);
563 Value *Op1 = I.getOperand(1);

578// Try to recognize below function as popcount intrinsic.
579// Ref. Hacker Delights
580// int popcount32(unsigned int i) {
581// uWord = (uWord & 0x55555555) + ((uWord>>1) & 0x55555555);
582// uWord = (uWord & 0x33333333) + ((uWord>>2) & 0x33333333);
583// uWord = (uWord & 0x0F0F0F0F) + ((uWord>>4) & 0x0F0F0F0F);
584// uWord = (uWord & 0x00FF00FF) + ((uWord>>8) & 0x00FF00FF);
585// return (uWord & 0x0000FFFF) + (uWord>>16);
586// }
587// int popcount64(unsigned long i) {
588// uWord = (uWord & 0x5555555555555555) + ((uWord>>1) & 0x5555555555555555);
589// uWord = (uWord & 0x3333333333333333) + ((uWord>>2) & 0x3333333333333333);
590// uWord = (uWord & 0x0F0F0F0F0F0F0F0F) + ((uWord>>4) & 0x0F0F0F0F0F0F0F0F);
591// uWord = (uWord & 0x00FF00FF00FF00FF) + ((uWord>>8) & 0x00FF00FF00FF00FF);
592// uWord = (uWord & 0x0000FFFF0000FFFF) + ((uWord>>16) & 0x0000FFFF0000FFFF);
593// return (uWord & 0x00000000FFFFFFFF) + (uWord>>32) & 0x00000000FFFFFFFF;
594// }
595//
596// InstCombine may narrow AND masks when it can prove the removed bits are
597// known zero (e.g. 0x0F0F0F0F -> 0x07070707). We accept such narrowed masks
598// by checking they are subsets of the expected masks and verifying the missing
599// bits are known zero via MaskedValueIsZero.
600static bool tryToRecognizePopCount1(Instruction &I) {
601 if (I.getOpcode() != Instruction::Add)
602 return false;
603
604 Type *Ty = I.getType();
605 if (!Ty->isIntOrIntVectorTy())
606 return false;
607
608 unsigned Len = Ty->getScalarSizeInBits();
609 if (Len > 64 || Len <= 8 || Len % 8 != 0)
610 return false;
611
612 // Len should be a power of 2 for the loop to work correctly
613 if (!isPowerOf2_32(Len))
614 return false;
615
616 APInt Mask55 = APInt::getSplat(Len, APInt(8, 0x55));
617 APInt Mask33 = APInt::getSplat(Len, APInt(8, 0x33));
618
619 SimplifyQuery SQ(I.getDataLayout());
620
621 // Check if CapturedMask is a valid (possibly narrowed) version of
622 // ExpectedMask for the given Operand. Returns true if the masks match
623 // exactly, or if CapturedMask is a subset and the missing bits are
624 // known zero in the Operand.
625 auto isValidNarrowedMask = [&](const APInt &CapturedMask,
626 const APInt &ExpectedMask,
627 Value *Operand) -> bool {
628 if (CapturedMask == ExpectedMask)
629 return true;
630 if (!CapturedMask.isSubsetOf(ExpectedMask))
631 return false;
632 APInt NeededMask = ExpectedMask & ~CapturedMask;
633 return MaskedValueIsZero(Operand, NeededMask, SQ);
634 };
635
636 // For "(x & M) + ((x >> S) & M)" patterns, both AND masks may be narrowed.
637 // Require subsets of BaseMask and prove any implied missing bits are zero.
638 auto narrowAddPairMasksOk = [&](const APInt &BaseMask, unsigned ShiftAmt,
639 Value *Val, const APInt &AndMask1,
640 const APInt &AndMask2) -> bool {
641 if (!AndMask1.isSubsetOf(BaseMask) || !AndMask2.isSubsetOf(BaseMask))
642 return false;
643 APInt NeededShifted = (BaseMask & ~AndMask1).shl(ShiftAmt);
644 APInt NeededUnshifted = BaseMask & ~AndMask2;
645 APInt AllNeeded = NeededShifted | NeededUnshifted;
646 return AllNeeded.isZero() || MaskedValueIsZero(Val, AllNeeded, SQ);
647 };
648
649 Value *ShiftOp;
650 Value *Start = &I;
651 for (unsigned I = Len; I >= 8; I = I / 2) {
652 APInt Mask = APInt::getSplat(Len, APInt::getLowBitsSet(I, I / 2));
653 const APInt *AndMask1 = nullptr, *AndMask2 = nullptr;
654
655 // Matching "(uWord & Mask) + ((uWord>>I/2) & Mask)".
656 // Both masks might have been narrowed by InstCombine.
657 if (match(Start,
658 m_c_Add(m_And(m_LShr(m_Value(ShiftOp), m_SpecificInt(I / 2)),
659 m_APInt(AndMask1)),
660 m_And(m_Deferred(ShiftOp), m_APInt(AndMask2))))) {
661 if (!narrowAddPairMasksOk(Mask, I / 2, ShiftOp, *AndMask1, *AndMask2))
662 return false;
663 }
664 // Matching "(uWord & Mask) + (uWord>>I/2)".
665 // The mask might have been narrowed by InstCombine.
666 else if (match(Start,
667 m_c_Add(m_LShr(m_Value(ShiftOp), m_SpecificInt(I / 2)),
668 m_And(m_Deferred(ShiftOp), m_APInt(AndMask1))))) {
669 if (!isValidNarrowedMask(*AndMask1, Mask, ShiftOp))
670 return false;
671 } else
672 return false;
673 Start = ShiftOp;
674 }
675
676 // Matching "uWord = (uWord & Mask33) + ((uWord>>2) & Mask33)".
677 const APInt *AndMask1 = nullptr, *AndMask2 = nullptr;
678 if (!match(Start, m_c_Add(m_And(m_LShr(m_Value(ShiftOp), m_SpecificInt(2)),
679 m_APInt(AndMask1)),
680 m_And(m_Deferred(ShiftOp), m_APInt(AndMask2)))))
681 return false;
682 if (!narrowAddPairMasksOk(Mask33, 2, ShiftOp, *AndMask1, *AndMask2))
683 return false;
684
685 Start = ShiftOp;
686 Value *Root;
687 // Matching "uWord = (uWord & Mask55) + ((uWord>>1) & Mask55)".
688 AndMask1 = nullptr;
689 AndMask2 = nullptr;
690 if (!match(Start, m_c_Add(m_And(m_LShr(m_Value(Root), m_SpecificInt(1)),
691 m_APInt(AndMask1)),
692 m_And(m_Deferred(Root), m_APInt(AndMask2)))))
693 return false;
694 if (!narrowAddPairMasksOk(Mask55, 1, Root, *AndMask1, *AndMask2))
695 return false;
696
697 replaceWithPopCount(I, Root);
698 return true;
699}

701// Try to recognize below function as popcount intrinsic.
702// Ref. Hackers Delight
703// int popcnt(unsigned x) {
704// x = x - ((x >> 1) & 0x55555555);
705// x = (x & 0x33333333) + ((x >> 2) & 0x33333333);
706// x = (x + (x >> 4)) & 0x0F0F0F0F;
707// x = x + (x >> 8);
708// x = x + (x >> 16);
709// return x & 0x0000003F;
710// }
711
712// int popcnt(unsigned x) {
713// x = x - ((x >> 1) & 0x55555555);
714// x = x - 3*((x >> 2) & 0x33333333);
715// x = (x + (x >> 4)) & 0x0F0F0F0F;
716// x = x + (x >> 8);
717// x = x + (x >> 16);
718// return x & 0x0000003F;
719// }
720static bool tryToRecognizePopCount2n3(Instruction &I) {
721 if (I.getOpcode() != Instruction::And)
722 return false;
723
724 Type *Ty = I.getType();
725 if (!Ty->isIntOrIntVectorTy())
726 return false;
727
728 unsigned Len = Ty->getScalarSizeInBits();
729 Value *Add1;
730 if (Len == 8) {
731 // Special case for Len == 8, we only need to match the And at the end of
732 // matchPopCountBytes.
733 Add1 = &I;
734 } else {
735 const APInt *MaskRes;
736 if (!match(&I, m_And(m_Value(Add1), m_APInt(MaskRes))))
737 return false;
738
739 // Since `(trunc (and x, C))` might be canonicalized into `(and (trunc x),
740 // C)` we might loose the opportunity to recognize `(trunc (popcount y))`.
741 // The following block tries to capture such truncation, update `Len`, and
742 // append the truncation at the end of the emitting popcount, if there is
743 // any.
744 Value *TruncSrc;
745 if (match(Add1, m_OneUse(m_Trunc(m_Value(TruncSrc))))) {
746 Add1 = TruncSrc;
747 Len = Add1->getType()->getScalarSizeInBits();
748 }
749
750 if (Len > 64 || Len <= 8 || Len % 8 != 0)
751 return false;
752
753 // Len should be a power of 2 for the loop to work correctly
754 if (!isPowerOf2_32(Len))
755 return false;
756
757 // Number of bits needed to represent Len.
758 unsigned NumLenBits = Log2_32(Len) + 1;
759 // The "mask" here really only needs to fulfill two conditions:
760 // (1) All ones for the lower NumLenBits-bits
761 // (2) Zeros from bit 8 and onward.
762 // Condition (1) is straightforward. The reason behind condition
763 // (2) is that we don't care any 8-bit chunks but the first one
764 // in the original divide-and-conquer algorithm.
765 if (MaskRes->countTrailingOnes() < NumLenBits ||
766 MaskRes->getActiveBits() > 8)
767 return false;
768
769 for (unsigned I = Len; I >= 16; I = I / 2) {
770 Value *Add2;
771 // Matching "x = x + (x >> I/2)" for I-bit.
772 if (!match(Add1, m_c_Add(m_LShr(m_Value(Add2), m_SpecificInt(I / 2)),
773 m_Deferred(Add2))))
774 return false;
775 Add1 = Add2;
776 }
777 }
778
779 Value *Root = matchPopCountBytes(Add1, Len, I.getDataLayout());
780 if (!Root)
781 return false;
782
783 replaceWithPopCount(I, Root);
784 return true;
785}

787/// Fold smin(smax(fptosi(x), C1), C2) to llvm.fptosi.sat(x), providing C1 and
788/// C2 saturate the value of the fp conversion. The transform is not reversable
789/// as the fptosi.sat is more defined than the input - all values produce a
790/// valid value for the fptosi.sat, where as some produce poison for original
791/// that were out of range of the integer conversion. The reversed pattern may
792/// use fmax and fmin instead. As we cannot directly reverse the transform, and
793/// it is not always profitable, we make it conditional on the cost being
794/// reported as lower by TTI.
795static bool tryToFPToSat(Instruction &I, TargetTransformInfo &TTI) {
796 // Look for min(max(fptosi, converting to fptosi_sat.
797 Value *In;
798 const APInt *MinC, *MaxC;
799 if (!match(&I, m_SMax(m_OneUse(m_SMin(m_OneUse(m_FPToSI(m_Value(In))),
800 m_APInt(MinC))),
801 m_APInt(MaxC))) &&
802 !match(&I, m_SMin(m_OneUse(m_SMax(m_OneUse(m_FPToSI(m_Value(In))),
803 m_APInt(MaxC))),
804 m_APInt(MinC))))
805 return false;
806
807 // Check that the constants clamp a saturate.
808 if (!(*MinC + 1).isPowerOf2() || -*MaxC != *MinC + 1)
809 return false;
810
811 Type *IntTy = I.getType();
812 Type *FpTy = In->getType();
813 Type *SatTy =
814 IntegerType::get(IntTy->getContext(), (*MinC + 1).exactLogBase2() + 1);
815 if (auto *VecTy = dyn_cast<VectorType>(IntTy))
816 SatTy = VectorType::get(SatTy, VecTy->getElementCount());
817
818 // Get the cost of the intrinsic, and check that against the cost of
819 // fptosi+smin+smax
820 InstructionCost SatCost = TTI.getIntrinsicInstrCost(
821 IntrinsicCostAttributes(Intrinsic::fptosi_sat, SatTy, {In}, {FpTy}),
822 TTI::TCK_RecipThroughput);
823 SatCost += TTI.getCastInstrCost(Instruction::SExt, IntTy, SatTy,
824 TTI::CastContextHint::None,
825 TTI::TCK_RecipThroughput);
826
827 InstructionCost MinMaxCost = TTI.getCastInstrCost(
828 Instruction::FPToSI, IntTy, FpTy, TTI::CastContextHint::None,
829 TTI::TCK_RecipThroughput);
830 MinMaxCost += TTI.getIntrinsicInstrCost(
831 IntrinsicCostAttributes(Intrinsic::smin, IntTy, {IntTy}),
832 TTI::TCK_RecipThroughput);
833 MinMaxCost += TTI.getIntrinsicInstrCost(
834 IntrinsicCostAttributes(Intrinsic::smax, IntTy, {IntTy}),
835 TTI::TCK_RecipThroughput);
836
837 if (SatCost >= MinMaxCost)
838 return false;
839
840 IRBuilder<> Builder(&I);
841 Value *Sat =
842 Builder.CreateIntrinsic(Intrinsic::fptosi_sat, {SatTy, FpTy}, In);
843 I.replaceAllUsesWith(Builder.CreateSExt(Sat, IntTy));
844 return true;
845}

847/// Try to replace a mathlib call to sqrt with the LLVM intrinsic. This avoids
848/// pessimistic codegen that has to account for setting errno and can enable
849/// vectorization.
850static bool foldSqrt(CallInst *Call, LibFunc Func, TargetTransformInfo &TTI,
851 TargetLibraryInfo &TLI, AssumptionCache &AC,
852 DominatorTree &DT) {
853 // If (1) this is a sqrt libcall, (2) we can assume that NAN is not created
854 // (because NNAN or the operand arg must not be less than -0.0) and (2) we
855 // would not end up lowering to a libcall anyway (which could change the value
856 // of errno), then:
857 // (1) errno won't be set.
858 // (2) it is safe to convert this to an intrinsic call.
859 Type *Ty = Call->getType();
860 Value *Arg = Call->getArgOperand(0);
861 if (TTI.haveFastSqrt(Ty) &&
862 (Call->hasNoNaNs() ||
863 cannotBeOrderedLessThanZero(
864 Arg, SimplifyQuery(Call->getDataLayout(), &TLI, &DT, &AC, Call)))) {
865 IRBuilder<> Builder(Call);
866 Value *NewSqrt =
867 Builder.CreateIntrinsic(Intrinsic::sqrt, Ty, Arg, Call, "sqrt");
868 Call->replaceAllUsesWith(NewSqrt);
869
870 // Explicitly erase the old call because a call with side effects is not
871 // trivially dead.
872 Call->eraseFromParent();
873 return true;
874 }
875
876 return false;
877}

898// Try to recognize table-based ctz implementation.
899// E.g., an example in C (for more cases please see the llvm/tests):
900// int f(unsigned x) {
901// static const char table[32] =
902// {0, 1, 28, 2, 29, 14, 24, 3, 30,
903// 22, 20, 15, 25, 17, 4, 8, 31, 27,
904// 13, 23, 21, 19, 16, 7, 26, 12, 18, 6, 11, 5, 10, 9};
905// return table[((unsigned)((x & -x) * 0x077CB531U)) >> 27];
906// }
907// this can be lowered to `cttz` instruction.
908// There is also a special case when the element is 0.
909//
910// The (x & -x) sets the lowest non-zero bit to 1. The multiply is a de-bruijn
911// sequence that contains each pattern of bits in it. The shift extracts
912// the top bits after the multiply, and that index into the table should
913// represent the number of trailing zeros in the original number.
914//
915// Here are some examples or LLVM IR for a 64-bit target:
916//
917// CASE 1:
918// %sub = sub i32 0, %x
919// %and = and i32 %sub, %x
920// %mul = mul i32 %and, 125613361
921// %shr = lshr i32 %mul, 27
922// %idxprom = zext i32 %shr to i64
923// %arrayidx = getelementptr inbounds [32 x i8], [32 x i8]* @ctz1.table, i64 0,
924// i64 %idxprom
925// %0 = load i8, i8* %arrayidx, align 1, !tbaa !8
926//
927// CASE 2:
928// %sub = sub i32 0, %x
929// %and = and i32 %sub, %x
930// %mul = mul i32 %and, 72416175
931// %shr = lshr i32 %mul, 26
932// %idxprom = zext i32 %shr to i64
933// %arrayidx = getelementptr inbounds [64 x i16], [64 x i16]* @ctz2.table,
934// i64 0, i64 %idxprom
935// %0 = load i16, i16* %arrayidx, align 2, !tbaa !8
936//
937// CASE 3:
938// %sub = sub i32 0, %x
939// %and = and i32 %sub, %x
940// %mul = mul i32 %and, 81224991
941// %shr = lshr i32 %mul, 27
942// %idxprom = zext i32 %shr to i64
943// %arrayidx = getelementptr inbounds [32 x i32], [32 x i32]* @ctz3.table,
944// i64 0, i64 %idxprom
945// %0 = load i32, i32* %arrayidx, align 4, !tbaa !8
946//
947// CASE 4:
948// %sub = sub i64 0, %x
949// %and = and i64 %sub, %x
950// %mul = mul i64 %and, 283881067100198605
951// %shr = lshr i64 %mul, 58
952// %arrayidx = getelementptr inbounds [64 x i8], [64 x i8]* @table, i64 0,
953// i64 %shr
954// %0 = load i8, i8* %arrayidx, align 1, !tbaa !8
955//
956// All these can be lowered to @llvm.cttz.i32/64 intrinsics.
957//
958// This shares its initial match (load from a GEP into a constant table with
959// a single variable index) with tryToRecognizeTableBasedLog2() below; see
960// tryToRecognizeTableBasedCttzOrLog2().
961static bool tryToRecognizeTableBasedCttz(LoadInst *LI, Type *AccessType,
962 GlobalVariable *GVTable, Value *GepIdx,
963 const APInt &GEPScale,
964 const DataLayout &DL) {
965 Value *X1;
966 const APInt *MulConst, *ShiftConst, *AndCst = nullptr;
967 // Check that the gep variable index is ((x & -x) * MulConst) >> ShiftConst.
968 // This might be extended to the pointer index type, and if the gep index type
969 // has been replaced with an i8 then a new And (and different ShiftConst) will
970 // be present.
971 auto MatchInner = m_LShr(
972 m_Mul(m_c_And(m_Neg(m_Value(X1)), m_Deferred(X1)), m_APInt(MulConst)),
973 m_APInt(ShiftConst));
974 if (!match(GepIdx, m_CastOrSelf(MatchInner)) &&
975 !match(GepIdx, m_CastOrSelf(m_And(MatchInner, m_APInt(AndCst)))))
976 return false;
977
978 unsigned InputBits = X1->getType()->getScalarSizeInBits();
979 if (InputBits != 16 && InputBits != 32 && InputBits != 64 && InputBits != 128)
980 return false;
981
982 if (!GEPScale.isIntN(InputBits) ||
983 !isCTTZTable(GVTable->getInitializer(), *MulConst, *ShiftConst,
984 AndCst ? *AndCst : APInt::getAllOnes(InputBits), AccessType,
985 InputBits, GEPScale.zextOrTrunc(InputBits), DL))
986 return false;
987
988 ConstantInt *ZeroTableElem = cast<ConstantInt>(
989 ConstantFoldLoadFromConst(GVTable->getInitializer(), AccessType, DL));
990 bool DefinedForZero = ZeroTableElem->equalsInt(InputBits);
991
992 IRBuilder<> B(LI);
993 ConstantInt *BoolConst = B.getInt1(!DefinedForZero);
994 Type *XType = X1->getType();
995 auto Cttz = B.CreateIntrinsic(Intrinsic::cttz, {XType}, {X1, BoolConst});
996 Value *Res = B.CreateZExtOrTrunc(Cttz, AccessType);
997
998 if (!DefinedForZero) {
999 // If the value in elem 0 isn't the same as InputBits, we still want to
1000 // produce the value from the table. Emit the select in AccessType with elem
1001 // 0 unchanged, as the table's element type may be wider than the input
1002 // type (and directly truncating ZeroTableElem into the input type could
1003 // incorrectly drop bits).
1004 auto Cmp = B.CreateICmpEQ(X1, ConstantInt::get(XType, 0));
1005 Res = B.CreateSelect(Cmp, ZeroTableElem, Res);
1006
1007 // The true branch of select handles the cttz(0) case, which is rare.
1008 if (!ProfcheckDisableMetadataFixes) {
1009 if (Instruction *SelectI = dyn_cast<Instruction>(Res))
1010 SelectI->setMetadata(
1011 LLVMContext::MD_prof,
1012 MDBuilder(SelectI->getContext()).createUnlikelyBranchWeights());
1013 }
1014
1015 // NOTE: If the table[0] is 0, but the cttz(0) is defined by the Target
1016 // it should be handled as: `cttz(x) & (typeSize - 1)`.
1017 }
1018
1019 LI->replaceAllUsesWith(Res);
1020
1021 return true;
1022}

1046// Try to recognize table-based log2 implementation.
1047// E.g., an example in C (for more cases please the llvm/tests):
1048// int f(unsigned v) {
1049// static const char table[32] =
1050// {0, 9, 1, 10, 13, 21, 2, 29, 11, 14, 16, 18, 22, 25, 3, 30,
1051// 8, 12, 20, 28, 15, 17, 24, 7, 19, 27, 23, 6, 26, 5, 4, 31};
1052//
1053// v |= v >> 1; // first round down to one less than a power of 2
1054// v |= v >> 2;
1055// v |= v >> 4;
1056// v |= v >> 8;
1057// v |= v >> 16;
1058//
1059// return table[(unsigned)(v * 0x07C4ACDDU) >> 27];
1060// }
1061// this can be lowered to `ctlz` instruction.
1062// There is also a special case when the element is 0.
1063//
1064// The >> and |= sequence sets all bits below the most significant set bit. The
1065// multiply is a de-bruijn sequence that contains each pattern of bits in it.
1066// The shift extracts the top bits after the multiply, and that index into the
1067// table should represent the floor log base 2 of the original number.
1068//
1069// Here are some examples of LLVM IR for a 64-bit target.
1070//
1071// CASE 1:
1072// %shr = lshr i32 %v, 1
1073// %or = or i32 %shr, %v
1074// %shr1 = lshr i32 %or, 2
1075// %or2 = or i32 %shr1, %or
1076// %shr3 = lshr i32 %or2, 4
1077// %or4 = or i32 %shr3, %or2
1078// %shr5 = lshr i32 %or4, 8
1079// %or6 = or i32 %shr5, %or4
1080// %shr7 = lshr i32 %or6, 16
1081// %or8 = or i32 %shr7, %or6
1082// %mul = mul i32 %or8, 130329821
1083// %shr9 = lshr i32 %mul, 27
1084// %idxprom = zext nneg i32 %shr9 to i64
1085// %arrayidx = getelementptr inbounds i8, ptr @table, i64 %idxprom
1086// %0 = load i8, ptr %arrayidx, align 1
1087//
1088// CASE 2:
1089// %shr = lshr i64 %v, 1
1090// %or = or i64 %shr, %v
1091// %shr1 = lshr i64 %or, 2
1092// %or2 = or i64 %shr1, %or
1093// %shr3 = lshr i64 %or2, 4
1094// %or4 = or i64 %shr3, %or2
1095// %shr5 = lshr i64 %or4, 8
1096// %or6 = or i64 %shr5, %or4
1097// %shr7 = lshr i64 %or6, 16
1098// %or8 = or i64 %shr7, %or6
1099// %shr9 = lshr i64 %or8, 32
1100// %or10 = or i64 %shr9, %or8
1101// %mul = mul i64 %or10, 285870213051386505
1102// %shr11 = lshr i64 %mul, 58
1103// %arrayidx = getelementptr inbounds i8, ptr @table, i64 %shr11
1104// %0 = load i8, ptr %arrayidx, align 1
1105//
1106// CASE 3:
1107// A variant where the most-significant set bit of the OR-cascade result is
1108// isolated via subtraction before the multiply, i.e.
1109// table[((v - (v >> 1)) * MulConst) >> ShiftConst], analogous to how the
1110// cttz pattern isolates the least-significant set bit via `x & -x`:
1111//
1112// %shr = lshr i64 %v, 1
1113// %or = or i64 %shr, %v
1114// ... (rest of the OR-cascade, as above) ...
1115// %shr11 = lshr i64 %or10, 1
1116// %sub = sub i64 %or10, %shr11
1117// %mul = mul i64 %sub, 571347909858961602
1118// %shr12 = lshr i64 %mul, 58
1119// %arrayidx = getelementptr inbounds i8, ptr @table, i64 %shr12
1120// %0 = load i8, ptr %arrayidx, align 1
1121//
1122// All these can be lowered to @llvm.ctlz.i32/64 intrinsics and a subtract.
1123//
1124// This shares its initial match (load from a GEP into a constant table with
1125// a single variable index) with tryToRecognizeTableBasedCttz() above; see
1126// tryToRecognizeTableBasedCttzOrLog2().
1127static bool tryToRecognizeTableBasedLog2(LoadInst *LI, Type *AccessType,
1128 GlobalVariable *GVTable, Value *GepIdx,
1129 const APInt &GEPScale,
1130 const DataLayout &DL,
1131 TargetTransformInfo &TTI) {
1132 Value *X;
1133 const APInt *MulConst, *ShiftConst;
1134 // Check that the gep variable index is (x * MulConst) >> ShiftConst.
1135 auto MatchInner =
1136 m_LShr(m_Mul(m_Value(X), m_APInt(MulConst)), m_APInt(ShiftConst));
1137 if (!match(GepIdx, m_CastOrSelf(MatchInner)))
1138 return false;
1139
1140 // The multiplied value may instead be the OR-cascade result with its
1141 // most-significant set bit isolated first via `v - (v >> 1)`: since every
1142 // bit below the MSB of an OR-cascade result is 1, this subtraction leaves
1143 // just the MSB, mirroring how tryToRecognizeTableBasedCttz() isolates the
1144 // least-significant set bit via `x & -x`.
1145 bool IsolatedMSB = false;
1146 Value *V;
1147 if (match(X, m_Sub(m_Value(V), m_LShr(m_Deferred(V), m_SpecificInt(1))))) {
1148 IsolatedMSB = true;
1149 X = V;
1150 }
1151
1152 unsigned InputBits = X->getType()->getScalarSizeInBits();
1153 if (InputBits != 16 && InputBits != 32 && InputBits != 64 && InputBits != 128)
1154 return false;
1155
1156 // Verify shift amount.
1157 // TODO: Allow other shift amounts when we have proper test coverage.
1158 if (*ShiftConst != InputBits - Log2_32(InputBits))
1159 return false;
1160
1161 // Match the sequence of OR operations with right shifts by powers of 2.
1162 for (unsigned ShiftAmt = InputBits / 2; ShiftAmt != 0; ShiftAmt /= 2) {
1163 Value *Y;
1164 if (!match(X, m_c_Or(m_LShr(m_Value(Y), m_SpecificInt(ShiftAmt)),
1165 m_Deferred(Y))))
1166 return false;
1167 X = Y;
1168 }
1169
1170 if (!GEPScale.isIntN(InputBits))
1171 return false;
1172
1173 if (IsolatedMSB) {
1174 // With the MSB isolated, the multiplicand for an input whose MSB is at bit
1175 // Idx is a single set bit rather than a run of low bits, which is exactly
1176 // what isCTTZTable() checks for (there is no additional masking here, so
1177 // pass an all-ones mask).
1178 if (!isCTTZTable(GVTable->getInitializer(), *MulConst, *ShiftConst,
1179 APInt::getAllOnes(InputBits), AccessType, InputBits,
1180 GEPScale.zextOrTrunc(InputBits), DL))
1181 return false;
1182 } else {
1183 if (!isLog2Table(GVTable->getInitializer(), *MulConst, *ShiftConst,
1184 AccessType, InputBits, GEPScale.zextOrTrunc(InputBits),
1185 DL))
1186 return false;
1187 }
1188
1189 ConstantInt *ZeroTableElem = cast<ConstantInt>(
1190 ConstantFoldLoadFromConst(GVTable->getInitializer(), AccessType, DL));
1191
1192 // Use InputBits - 1 - ctlz(X) to compute log2(X).
1193 IRBuilder<> B(LI);
1194 ConstantInt *BoolConst = B.getTrue();
1195 Type *XType = X->getType();
1196
1197 // Check the the backend has an efficient ctlz instruction.
1198 // FIXME: Teach the backend to emit the original code when ctlz isn't
1199 // supported like we do for cttz.
1200 IntrinsicCostAttributes Attrs(
1201 Intrinsic::ctlz, XType,
1202 {PoisonValue::get(XType), /*is_zero_poison=*/BoolConst});
1203 InstructionCost Cost =
1204 TTI.getIntrinsicInstrCost(Attrs, TargetTransformInfo::TCK_SizeAndLatency);
1205 if (Cost > TargetTransformInfo::TCC_Basic)
1206 return false;
1207
1208 Value *Ctlz = B.CreateIntrinsic(Intrinsic::ctlz, {XType}, {X, BoolConst});
1209
1210 Constant *InputBitsM1 = ConstantInt::get(XType, InputBits - 1);
1211 Value *Sub = B.CreateSub(InputBitsM1, Ctlz);
1212
1213 // The table won't produce a sensible result for 0.
1214 Value *Cmp = B.CreateICmpEQ(X, ConstantInt::get(XType, 0));
1215 Value *Select = B.CreateSelect(Cmp, B.CreateZExt(ZeroTableElem, XType), Sub);
1216
1217 // The true branch of select handles the log2(0) case, which is rare.
1218 if (!ProfcheckDisableMetadataFixes) {
1219 if (Instruction *SelectI = dyn_cast<Instruction>(Select))
1220 SelectI->setMetadata(
1221 LLVMContext::MD_prof,
1222 MDBuilder(SelectI->getContext()).createUnlikelyBranchWeights());
1223 }
1224
1225 Value *ZExtOrTrunc = B.CreateZExtOrTrunc(Select, AccessType);
1226
1227 LI->replaceAllUsesWith(ZExtOrTrunc);
1228
1229 return true;
1230}

1232// Match a table-based cttz or log2 implementation. These patterns share a
1233// load from a global table pattern that we match first. Then we try the
1234// specific matches for the cttz and log2 patterns.
1235static bool tryToRecognizeTableBasedCttzOrLog2(Instruction &I,
1236 const DataLayout &DL,
1237 TargetTransformInfo &TTI) {
1238 LoadInst *LI = dyn_cast<LoadInst>(&I);
1239 if (!LI)
1240 return false;
1241
1242 Type *AccessType = LI->getType();
1243 if (!AccessType->isIntegerTy())
1244 return false;
1245
1246 GetElementPtrInst *GEP = dyn_cast<GetElementPtrInst>(LI->getPointerOperand());
1247 if (!GEP || !GEP->hasNoUnsignedSignedWrap())
1248 return false;
1249
1250 GlobalVariable *GVTable = dyn_cast<GlobalVariable>(GEP->getPointerOperand());
1251 if (!GVTable || !GVTable->hasInitializer() || !GVTable->isConstant())
1252 return false;
1253
1254 unsigned BW = DL.getIndexTypeSizeInBits(GEP->getType());
1255 APInt ModOffset(BW, 0);
1256 SmallMapVector<Value *, APInt, 4> VarOffsets;
1257 if (!GEP->collectOffset(DL, BW, VarOffsets, ModOffset) ||
1258 VarOffsets.size() != 1 || ModOffset != 0)
1259 return false;
1260 auto [GepIdx, GEPScale] = VarOffsets.front();
1261
1262 if (tryToRecognizeTableBasedCttz(LI, AccessType, GVTable, GepIdx, GEPScale,
1263 DL))
1264 return true;
1265
1266 return tryToRecognizeTableBasedLog2(LI, AccessType, GVTable, GepIdx, GEPScale,
1267 DL, TTI);
1268}

1270/// This is used by foldLoadsRecursive() to capture a Root Load node which is
1271/// of type or(load, load) and recursively build the wide load. Also capture the
1272/// shift amount, zero extend type and loadSize.
1273struct LoadOps {
1274 LoadInst *Root = nullptr;
1275 LoadInst *RootInsert = nullptr;
1276 bool FoundRoot = false;
1277 uint64_t LoadSize = 0;
1278 uint64_t Shift = 0;
1279 Type *ZextType;
1280 AAMDNodes AATags;
1281};
1282
1283// Identify and Merge consecutive loads recursively which is of the form
1284// (ZExt(L1) << shift1) | (ZExt(L2) << shift2) -> ZExt(L3) << shift1
1285// (ZExt(L1) << shift1) | ZExt(L2) -> ZExt(L3)
1286static bool foldLoadsRecursive(Value *V, LoadOps &LOps, const DataLayout &DL,
1287 AliasAnalysis &AA, bool IsRoot = false) {
1288 uint64_t ShAmt2;
1289 Value *X;
1290 Instruction *L1, *L2;
1291
1292 // For the root instruction, allow multiple uses since the final result
1293 // may legitimately be used in multiple places. For intermediate values,
1294 // require single use to avoid creating duplicate loads.
1295 if (!IsRoot && !V->hasOneUse())
1296 return false;
1297
1298 if (!match(V, m_c_Or(m_Value(X),
1299 m_OneUse(m_ShlOrSelf(m_OneUse(m_ZExt(m_Instruction(L2))),
1300 ShAmt2)))))
1301 return false;
1302
1303 if (!foldLoadsRecursive(X, LOps, DL, AA, /*IsRoot=*/false) && LOps.FoundRoot)
1304 // Avoid Partial chain merge.
1305 return false;
1306
1307 // Check if the pattern has loads
1308 LoadInst *LI1 = LOps.Root;
1309 uint64_t ShAmt1 = LOps.Shift;
1310 if (LOps.FoundRoot == false &&
1311 match(X, m_OneUse(
1312 m_ShlOrSelf(m_OneUse(m_ZExt(m_Instruction(L1))), ShAmt1)))) {
1313 LI1 = dyn_cast<LoadInst>(L1);
1314 }
1315 LoadInst *LI2 = dyn_cast<LoadInst>(L2);
1316
1317 // Check if loads are same, atomic, volatile and having same address space.
1318 if (LI1 == LI2 || !LI1 || !LI2 || !LI1->isSimple() || !LI2->isSimple() ||
1319 LI1->getPointerAddressSpace() != LI2->getPointerAddressSpace())
1320 return false;
1321
1322 // Check if Loads come from same BB.
1323 if (LI1->getParent() != LI2->getParent())
1324 return false;
1325
1326 // Find the data layout
1327 bool IsBigEndian = DL.isBigEndian();
1328
1329 // Check if loads are consecutive and same size.
1330 Value *Load1Ptr = LI1->getPointerOperand();
1331 APInt Offset1(DL.getIndexTypeSizeInBits(Load1Ptr->getType()), 0);
1332 Load1Ptr =
1333 Load1Ptr->stripAndAccumulateConstantOffsets(DL, Offset1,
1334 /* AllowNonInbounds */ true);
1335
1336 Value *Load2Ptr = LI2->getPointerOperand();
1337 APInt Offset2(DL.getIndexTypeSizeInBits(Load2Ptr->getType()), 0);
1338 Load2Ptr =
1339 Load2Ptr->stripAndAccumulateConstantOffsets(DL, Offset2,
1340 /* AllowNonInbounds */ true);
1341
1342 // Verify if both loads have same base pointers
1343 uint64_t LoadSize1 = LI1->getType()->getPrimitiveSizeInBits();
1344 uint64_t LoadSize2 = LI2->getType()->getPrimitiveSizeInBits();
1345 if (Load1Ptr != Load2Ptr)
1346 return false;
1347
1348 // Make sure that there are no padding bits.
1349 if (!DL.typeSizeEqualsStoreSize(LI1->getType()) ||
1350 !DL.typeSizeEqualsStoreSize(LI2->getType()))
1351 return false;
1352
1353 // Alias Analysis to check for stores b/w the loads.
1354 LoadInst *Start = LOps.FoundRoot ? LOps.RootInsert : LI1, *End = LI2;
1355 MemoryLocation Loc;
1356 if (!Start->comesBefore(End)) {
1357 std::swap(Start, End);
1358 // If LOps.RootInsert comes after LI2, since we use LI2 as the new insert
1359 // point, we should make sure whether the memory region accessed by LOps
1360 // isn't modified.
1361 if (LOps.FoundRoot)
1362 Loc = MemoryLocation(
1363 LOps.Root->getPointerOperand(),
1364 LocationSize::precise(DL.getTypeStoreSize(
1365 IntegerType::get(LI1->getContext(), LOps.LoadSize))),
1366 LOps.AATags);
1367 else
1368 Loc = MemoryLocation::get(End);
1369 } else
1370 Loc = MemoryLocation::get(End);
1371 unsigned NumScanned = 0;
1372 for (Instruction &Inst :
1373 make_range(Start->getIterator(), End->getIterator())) {
1374 if (Inst.mayWriteToMemory() && isModSet(AA.getModRefInfo(&Inst, Loc)))
1375 return false;
1376
1377 if (++NumScanned > MaxInstrsToScan)
1378 return false;
1379 }
1380
1381 // Make sure Load with lower Offset is at LI1
1382 bool Reverse = false;
1383 if (Offset2.slt(Offset1)) {
1384 std::swap(LI1, LI2);
1385 std::swap(ShAmt1, ShAmt2);
1386 std::swap(Offset1, Offset2);
1387 std::swap(Load1Ptr, Load2Ptr);
1388 std::swap(LoadSize1, LoadSize2);
1389 Reverse = true;
1390 }
1391
1392 // Big endian swap the shifts
1393 if (IsBigEndian)
1394 std::swap(ShAmt1, ShAmt2);
1395
1396 // First load is always LI1. This is where we put the new load.
1397 // Use the merged load size available from LI1 for forward loads.
1398 if (LOps.FoundRoot) {
1399 if (!Reverse)
1400 LoadSize1 = LOps.LoadSize;
1401 else
1402 LoadSize2 = LOps.LoadSize;
1403 }
1404
1405 // Verify if shift amount and load index aligns and verifies that loads
1406 // are consecutive.
1407 uint64_t ShiftDiff = IsBigEndian ? LoadSize2 : LoadSize1;
1408 uint64_t PrevSize =
1409 DL.getTypeStoreSize(IntegerType::get(LI1->getContext(), LoadSize1));
1410 if ((ShAmt2 - ShAmt1) != ShiftDiff || (Offset2 - Offset1) != PrevSize)
1411 return false;
1412
1413 // Reject if the combined size of the loads exceeds the target type size.
1414 // This avoids attempting to emit an invalid ZExt (from wider to narrower
1415 // type) when out-of-bounds shifts lead to matching too many loads.
1416 if (LoadSize1 + LoadSize2 > X->getType()->getScalarSizeInBits())
1417 return false;
1418
1419 // Update LOps
1420 AAMDNodes AATags1 = LOps.AATags;
1421 AAMDNodes AATags2 = LI2->getAAMetadata();
1422 if (LOps.FoundRoot == false) {
1423 LOps.FoundRoot = true;
1424 AATags1 = LI1->getAAMetadata();
1425 }
1426 LOps.LoadSize = LoadSize1 + LoadSize2;
1427 LOps.RootInsert = Start;
1428
1429 // Concatenate the AATags of the Merged Loads.
1430 LOps.AATags = AATags1.concat(AATags2);
1431
1432 LOps.Root = LI1;
1433 LOps.Shift = ShAmt1;
1434 LOps.ZextType = X->getType();
1435 return true;
1436}

1438// For a given BB instruction, evaluate all loads in the chain that form a
1439// pattern which suggests that the loads can be combined. The one and only use
1440// of the loads is to form a wider load.
1441static bool foldConsecutiveLoads(Instruction &I, const DataLayout &DL,
1442 TargetTransformInfo &TTI, AliasAnalysis &AA,
1443 const DominatorTree &DT) {
1444 // Only consider load chains of scalar values.
1445 if (isa<VectorType>(I.getType()))
1446 return false;
1447
1448 LoadOps LOps;
1449 if (!foldLoadsRecursive(&I, LOps, DL, AA, /*IsRoot=*/true) || !LOps.FoundRoot)
1450 return false;
1451
1452 IRBuilder<> Builder(&I);
1453 LoadInst *NewLoad = nullptr, *LI1 = LOps.Root;
1454
1455 IntegerType *WiderType = IntegerType::get(I.getContext(), LOps.LoadSize);
1456 // TTI based checks if we want to proceed with wider load
1457 bool Allowed = TTI.isTypeLegal(WiderType);
1458 if (!Allowed)
1459 return false;
1460
1461 unsigned AS = LI1->getPointerAddressSpace();
1462 unsigned Fast = 0;
1463 Allowed = TTI.allowsMisalignedMemoryAccesses(I.getContext(), LOps.LoadSize,
1464 AS, LI1->getAlign(), &Fast);
1465 if (!Allowed || !Fast)
1466 return false;
1467
1468 // Get the Index and Ptr for the new GEP.
1469 Value *Load1Ptr = LI1->getPointerOperand();
1470 Builder.SetInsertPoint(LOps.RootInsert);
1471 if (!DT.dominates(Load1Ptr, LOps.RootInsert)) {
1472 APInt Offset1(DL.getIndexTypeSizeInBits(Load1Ptr->getType()), 0);
1473 Load1Ptr = Load1Ptr->stripAndAccumulateConstantOffsets(
1474 DL, Offset1, /* AllowNonInbounds */ true);
1475 Load1Ptr = Builder.CreatePtrAdd(Load1Ptr, Builder.getInt(Offset1));
1476 }
1477 // Generate wider load.
1478 NewLoad = Builder.CreateAlignedLoad(WiderType, Load1Ptr, LI1->getAlign(),
1479 LI1->isVolatile(), "");
1480 NewLoad->takeName(LI1);
1481 // Set the New Load AATags Metadata.
1482 if (LOps.AATags)
1483 NewLoad->setAAMetadata(LOps.AATags);
1484
1485 Value *NewOp = NewLoad;
1486 // Zero extend if needed.
1487 NewOp = Builder.CreateZExt(NewOp, LOps.ZextType);
1488
1489 // Check if shift needed. We need to shift with the amount of load1
1490 // shift if not zero.
1491 if (LOps.Shift)
1492 NewOp = Builder.CreateShl(NewOp, LOps.Shift);
1493 I.replaceAllUsesWith(NewOp);
1494
1495 return true;
1496}

1498/// ValWidth bits starting at ValOffset of Val stored at PtrBase+PtrOffset.
1499struct PartStore {
1500 Value *PtrBase;
1501 APInt PtrOffset;
1502 Value *Val;
1503 uint64_t ValOffset;
1504 uint64_t ValWidth;
1505 StoreInst *Store;
1506
1507 bool isCompatibleWith(const PartStore &Other) const {
1508 return PtrBase == Other.PtrBase && Val == Other.Val;
1509 }
1510
1511 bool operator<(const PartStore &Other) const {
1512 return PtrOffset.slt(Other.PtrOffset);
1513 }
1514};
1515
1516static std::optional<PartStore> matchPartStore(Instruction &I,
1517 const DataLayout &DL) {
1518 auto *Store = dyn_cast<StoreInst>(&I);
1519 if (!Store || !Store->isSimple())
1520 return std::nullopt;
1521
1522 Value *StoredVal = Store->getValueOperand();
1523 Type *StoredTy = StoredVal->getType();
1524 if (!StoredTy->isIntegerTy() || !DL.typeSizeEqualsStoreSize(StoredTy))
1525 return std::nullopt;
1526
1527 uint64_t ValWidth = StoredTy->getPrimitiveSizeInBits();
1528 uint64_t ValOffset;
1529 Value *Val;
1530 if (!match(StoredVal, m_Trunc(m_LShrOrSelf(m_Value(Val), ValOffset))))
1531 return std::nullopt;
1532
1533 Value *Ptr = Store->getPointerOperand();
1534 APInt PtrOffset(DL.getIndexTypeSizeInBits(Ptr->getType()), 0);
1535 Value *PtrBase = Ptr->stripAndAccumulateConstantOffsets(
1536 DL, PtrOffset, /*AllowNonInbounds=*/true);
1537 return {{PtrBase, PtrOffset, Val, ValOffset, ValWidth, Store}};
1538}

1540static bool mergeConsecutivePartStores(ArrayRef<PartStore> Parts,
1541 unsigned Width, const DataLayout &DL,
1542 TargetTransformInfo &TTI) {
1543 if (Parts.size() < 2)
1544 return false;
1545
1546 // Check whether combining the stores is profitable.
1547 // FIXME: We could generate smaller stores if we can't produce a large one.
1548 const PartStore &First = Parts.front();
1549 LLVMContext &Ctx = First.Store->getContext();
1550 Type *NewTy = Type::getIntNTy(Ctx, Width);
1551 unsigned Fast = 0;
1552 if (!TTI.isTypeLegal(NewTy) ||
1553 !TTI.allowsMisalignedMemoryAccesses(Ctx, Width,
1554 First.Store->getPointerAddressSpace(),
1555 First.Store->getAlign(), &Fast) ||
1556 !Fast)
1557 return false;
1558
1559 // Generate the combined store.
1560 IRBuilder<> Builder(First.Store);
1561 Value *Val = First.Val;
1562 if (First.ValOffset != 0)
1563 Val = Builder.CreateLShr(Val, First.ValOffset);
1564 Val = Builder.CreateZExtOrTrunc(Val, NewTy);
1565 StoreInst *Store = Builder.CreateAlignedStore(
1566 Val, First.Store->getPointerOperand(), First.Store->getAlign());
1567
1568 // Merge various metadata onto the new store.
1569 AAMDNodes AATags = First.Store->getAAMetadata();
1570 SmallVector<Instruction *> Stores = {First.Store};
1571 Stores.reserve(Parts.size());
1572 SmallVector<DebugLoc> DbgLocs = {First.Store->getDebugLoc()};
1573 DbgLocs.reserve(Parts.size());
1574 for (const PartStore &Part : drop_begin(Parts)) {
1575 AATags = AATags.concat(Part.Store->getAAMetadata());
1576 Stores.push_back(Part.Store);
1577 DbgLocs.push_back(Part.Store->getDebugLoc());
1578 }
1579 Store->setAAMetadata(AATags);
1580 Store->mergeDIAssignID(Stores);
1581 Store->setDebugLoc(DebugLoc::getMergedLocations(DbgLocs));
1582
1583 // Remove the old stores.
1584 for (const PartStore &Part : Parts)
1585 Part.Store->eraseFromParent();
1586
1587 return true;
1588}
1589
1590static bool mergePartStores(SmallVectorImpl<PartStore> &Parts,
1591 const DataLayout &DL, TargetTransformInfo &TTI) {
1592 if (Parts.size() < 2)
1593 return false;
1594
1595 // We now have multiple parts of the same value stored to the same pointer.
1596 // Sort the parts by pointer offset, and make sure they are consistent with
1597 // the value offsets. Also check that the value is fully covered without
1598 // overlaps.
1599 bool Changed = false;
1600 llvm::sort(Parts);
1601 int64_t LastEndOffsetFromFirst = 0;
1602 const PartStore *First = &Parts[0];
1603 for (const PartStore &Part : Parts) {
1604 APInt PtrOffsetFromFirst = Part.PtrOffset - First->PtrOffset;
1605 int64_t ValOffsetFromFirst = Part.ValOffset - First->ValOffset;
1606 if (PtrOffsetFromFirst * 8 != ValOffsetFromFirst ||
1607 LastEndOffsetFromFirst != ValOffsetFromFirst) {
1608 Changed |= mergeConsecutivePartStores(ArrayRef(First, &Part),
1609 LastEndOffsetFromFirst, DL, TTI);
1610 First = &Part;
1611 LastEndOffsetFromFirst = Part.ValWidth;
1612 continue;
1613 }
1614
1615 LastEndOffsetFromFirst = ValOffsetFromFirst + Part.ValWidth;
1616 }
1617
1618 Changed |= mergeConsecutivePartStores(ArrayRef(First, Parts.end()),
1619 LastEndOffsetFromFirst, DL, TTI);
1620 return Changed;
1621}

1623static bool foldConsecutiveStores(BasicBlock &BB, const DataLayout &DL,
1624 TargetTransformInfo &TTI, AliasAnalysis &AA) {
1625 // FIXME: Add big endian support.
1626 if (DL.isBigEndian())
1627 return false;
1628
1629 BatchAAResults BatchAA(AA);
1630 SmallVector<PartStore, 8> Parts;
1631 bool MadeChange = false;
1632 for (Instruction &I : make_early_inc_range(BB)) {
1633 if (std::optional<PartStore> Part = matchPartStore(I, DL)) {
1634 if (Parts.empty() || Part->isCompatibleWith(Parts[0])) {
1635 Parts.push_back(std::move(*Part));
1636 continue;
1637 }
1638
1639 MadeChange |= mergePartStores(Parts, DL, TTI);
1640 Parts.clear();
1641 Parts.push_back(std::move(*Part));
1642 continue;
1643 }
1644
1645 if (Parts.empty())
1646 continue;
1647
1648 if (I.mayThrow() ||
1649 (I.mayReadOrWriteMemory() &&
1650 isModOrRefSet(BatchAA.getModRefInfo(
1651 &I, MemoryLocation::getBeforeOrAfter(Parts[0].PtrBase))))) {
1652 MadeChange |= mergePartStores(Parts, DL, TTI);
1653 Parts.clear();
1654 continue;
1655 }
1656 }
1657
1658 MadeChange |= mergePartStores(Parts, DL, TTI);
1659 return MadeChange;
1660}

1662/// Combine away instructions providing they are still equivalent when compared
1663/// against 0. i.e do they have any bits set.
1664static Value *optimizeShiftInOrChain(Value *V, IRBuilder<> &Builder) {
1665 auto *I = dyn_cast<Instruction>(V);
1666 if (!I || I->getOpcode() != Instruction::Or || !I->hasOneUse())
1667 return nullptr;
1668
1669 Value *A;
1670
1671 // Look deeper into the chain of or's, combining away shl (so long as they are
1672 // nuw or nsw).
1673 Value *Op0 = I->getOperand(0);
1674 if (match(Op0, m_CombineOr(m_NSWShl(m_Value(A), m_Value()),
1675 m_NUWShl(m_Value(A), m_Value()))))
1676 Op0 = A;
1677 else if (auto *NOp = optimizeShiftInOrChain(Op0, Builder))
1678 Op0 = NOp;
1679
1680 Value *Op1 = I->getOperand(1);
1681 if (match(Op1, m_CombineOr(m_NSWShl(m_Value(A), m_Value()),
1682 m_NUWShl(m_Value(A), m_Value()))))
1683 Op1 = A;
1684 else if (auto *NOp = optimizeShiftInOrChain(Op1, Builder))
1685 Op1 = NOp;
1686
1687 if (Op0 != I->getOperand(0) || Op1 != I->getOperand(1))
1688 return Builder.CreateOr(Op0, Op1);
1689 return nullptr;
1690}

1692static bool foldICmpOrChain(Instruction &I, const DataLayout &DL,
1693 TargetTransformInfo &TTI, AliasAnalysis &AA,
1694 const DominatorTree &DT) {
1695 CmpPredicate Pred;
1696 Value *Op0;
1697 if (!match(&I, m_ICmp(Pred, m_Value(Op0), m_Zero())) ||
1698 !ICmpInst::isEquality(Pred))
1699 return false;
1700
1701 // If the chain or or's matches a load, combine to that before attempting to
1702 // remove shifts.
1703 if (auto OpI = dyn_cast<Instruction>(Op0))
1704 if (OpI->getOpcode() == Instruction::Or)
1705 if (foldConsecutiveLoads(*OpI, DL, TTI, AA, DT))
1706 return true;
1707
1708 IRBuilder<> Builder(&I);
1709 // icmp eq/ne or(shl(a), b), 0 -> icmp eq/ne or(a, b), 0
1710 if (auto *Res = optimizeShiftInOrChain(Op0, Builder)) {
1711 I.replaceAllUsesWith(Builder.CreateICmp(Pred, Res, I.getOperand(1)));
1712 return true;
1713 }
1714
1715 return false;
1716}

1718// Calculate GEP Stride and accumulated const ModOffset. Return Stride and
1719// ModOffset
1720static std::pair<APInt, APInt>
1721getStrideAndModOffsetOfGEP(Value *PtrOp, const DataLayout &DL) {
1722 unsigned BW = DL.getIndexTypeSizeInBits(PtrOp->getType());
1723 std::optional<APInt> Stride;
1724 APInt ModOffset(BW, 0);
1725 // Return a minimum gep stride, greatest common divisor of consective gep
1726 // index scales(c.f. Bézout's identity).
1727 while (auto *GEP = dyn_cast<GEPOperator>(PtrOp)) {
1728 SmallMapVector<Value *, APInt, 4> VarOffsets;
1729 if (!GEP->collectOffset(DL, BW, VarOffsets, ModOffset))
1730 break;
1731
1732 for (auto [V, Scale] : VarOffsets) {
1733 // Only keep a power of two factor for non-inbounds
1734 if (!GEP->hasNoUnsignedSignedWrap())
1735 Scale = APInt::getOneBitSet(Scale.getBitWidth(), Scale.countr_zero());
1736
1737 if (!Stride)
1738 Stride = Scale;
1739 else
1740 Stride = APIntOps::GreatestCommonDivisor(*Stride, Scale);
1741 }
1742
1743 PtrOp = GEP->getPointerOperand();
1744 }
1745
1746 // Check whether pointer arrives back at Global Variable via at least one GEP.
1747 // Even if it doesn't, we can check by alignment.
1748 if (!isa<GlobalVariable>(PtrOp) || !Stride)
1749 return {APInt(BW, 1), APInt(BW, 0)};
1750
1751 // In consideration of signed GEP indices, non-negligible offset become
1752 // remainder of division by minimum GEP stride.
1753 ModOffset = ModOffset.srem(*Stride);
1754 if (ModOffset.isNegative())
1755 ModOffset += *Stride;
1756
1757 return {*Stride, ModOffset};
1758}

1760/// If C is a constant patterned array and all valid loaded results for given
1761/// alignment are same to a constant, return that constant.
1762static bool foldPatternedLoads(Instruction &I, const DataLayout &DL) {
1763 auto *LI = dyn_cast<LoadInst>(&I);
1764 if (!LI || LI->isVolatile())
1765 return false;
1766
1767 // We can only fold the load if it is from a constant global with definitive
1768 // initializer. Skip expensive logic if this is not the case.
1769 auto *PtrOp = LI->getPointerOperand();
1770 auto *GV = dyn_cast<GlobalVariable>(getUnderlyingObject(PtrOp));
1771 if (!GV || !GV->isConstant() || !GV->hasDefinitiveInitializer())
1772 return false;
1773
1774 // Bail for large initializers in excess of 4K to avoid too many scans.
1775 Constant *C = GV->getInitializer();
1776 uint64_t GVSize = DL.getTypeAllocSize(C->getType());
1777 if (!GVSize || 4096 < GVSize)
1778 return false;
1779
1780 Type *LoadTy = LI->getType();
1781 unsigned BW = DL.getIndexTypeSizeInBits(PtrOp->getType());
1782 auto [Stride, ConstOffset] = getStrideAndModOffsetOfGEP(PtrOp, DL);
1783
1784 // Any possible offset could be multiple of GEP stride. And any valid
1785 // offset is multiple of load alignment, so checking only multiples of bigger
1786 // one is sufficient to say results' equality.
1787 if (auto LA = LI->getAlign();
1788 LA <= GV->getAlign().valueOrOne() && Stride.getZExtValue() < LA.value()) {
1789 ConstOffset = APInt(BW, 0);
1790 Stride = APInt(BW, LA.value());
1791 }
1792
1793 Constant *Ca = ConstantFoldLoadFromConst(C, LoadTy, ConstOffset, DL);
1794 if (!Ca)
1795 return false;
1796
1797 unsigned E = GVSize - DL.getTypeStoreSize(LoadTy);
1798 for (; ConstOffset.getZExtValue() <= E; ConstOffset += Stride)
1799 if (Ca != ConstantFoldLoadFromConst(C, LoadTy, ConstOffset, DL))
1800 return false;
1801
1802 I.replaceAllUsesWith(Ca);
1803
1804 return true;
1805}

1827/// First we normalize calls to strncmp/strcmp to the form of
1828/// compare(s1, s2, N), which means comparing first N bytes of s1 and s2
1829/// (without considering '\0').
1830///
1831/// Examples:
1832///
1833/// \code
1834/// strncmp(s, "a", 3) -> compare(s, "a", 2)
1835/// strncmp(s, "abc", 3) -> compare(s, "abc", 3)
1836/// strncmp(s, "a\0b", 3) -> compare(s, "a\0b", 2)
1837/// strcmp(s, "a") -> compare(s, "a", 2)
1838///
1839/// char s2[] = {'a'}
1840/// strncmp(s, s2, 3) -> compare(s, s2, 3)
1841///
1842/// char s2[] = {'a', 'b', 'c', 'd'}
1843/// strncmp(s, s2, 3) -> compare(s, s2, 3)
1844/// \endcode
1845///
1846/// We only handle cases where N and exactly one of s1 and s2 are constant.
1847/// Cases that s1 and s2 are both constant are already handled by the
1848/// instcombine pass.
1849///
1850/// We do not handle cases where N > StrNCmpInlineThreshold.
1851///
1852/// We also do not handles cases where N < 2, which are already
1853/// handled by the instcombine pass.
1854///
1855bool StrNCmpInliner::optimizeStrNCmp() {
1856 if (StrNCmpInlineThreshold < 2)
1857 return false;
1858
1859 if (!isOnlyUsedInZeroComparison(CI))
1860 return false;
1861
1862 Value *Str1P = CI->getArgOperand(0);
1863 Value *Str2P = CI->getArgOperand(1);
1864 // Should be handled elsewhere.
1865 if (Str1P == Str2P)
1866 return false;
1867
1868 StringRef Str1, Str2;
1869 bool HasStr1 = getConstantStringInfo(Str1P, Str1, /*TrimAtNul=*/false);
1870 bool HasStr2 = getConstantStringInfo(Str2P, Str2, /*TrimAtNul=*/false);
1871 if (HasStr1 == HasStr2)
1872 return false;
1873
1874 // Note that '\0' and characters after it are not trimmed.
1875 StringRef Str = HasStr1 ? Str1 : Str2;
1876 Value *StrP = HasStr1 ? Str2P : Str1P;
1877
1878 size_t Idx = Str.find('\0');
1879 uint64_t N = Idx == StringRef::npos ? UINT64_MAX : Idx + 1;
1880 if (Func == LibFunc_strncmp) {
1881 if (auto *ConstInt = dyn_cast<ConstantInt>(CI->getArgOperand(2)))
1882 N = std::min(N, ConstInt->getZExtValue());
1883 else
1884 return false;
1885 }
1886 // Now N means how many bytes we need to compare at most.
1887 if (N > Str.size() || N < 2 || N > StrNCmpInlineThreshold)
1888 return false;
1889
1890 // Cases where StrP has two or more dereferenceable bytes might be better
1891 // optimized elsewhere.
1892 bool CanBeNull = false;
1893 if (StrP->getPointerDereferenceableBytes(DL, CanBeNull,
1894 /*CanBeFreed=*/nullptr) > 1)
1895 return false;
1896 inlineCompare(StrP, Str, N, HasStr1);
1897 return true;
1898}

1900/// Convert
1901///
1902/// \code
1903/// ret = compare(s1, s2, N)
1904/// \endcode
1905///
1906/// into
1907///
1908/// \code
1909/// ret = (int)s1[0] - (int)s2[0]
1910/// if (ret != 0)
1911/// goto NE
1912/// ...
1913/// ret = (int)s1[N-2] - (int)s2[N-2]
1914/// if (ret != 0)
1915/// goto NE
1916/// ret = (int)s1[N-1] - (int)s2[N-1]
1917/// NE:
1918/// \endcode
1919///
1920/// CFG before and after the transformation:
1921///
1922/// (before)
1923/// BBCI
1924///
1925/// (after)
1926/// BBCI -> BBSubs[0] (sub,icmp) --NE-> BBNE -> BBTail
1927/// | ^
1928/// E |
1929/// | |
1930/// BBSubs[1] (sub,icmp) --NE-----+
1931/// ... |
1932/// BBSubs[N-1] (sub) ---------+
1933///
1934void StrNCmpInliner::inlineCompare(Value *LHS, StringRef RHS, uint64_t N,
1935 bool Swapped) {
1936 auto &Ctx = CI->getContext();
1937 IRBuilder<> B(Ctx);
1938 // We want these instructions to be recognized as inlined instructions for the
1939 // compare call, but we don't have a source location for the definition of
1940 // that function, since we're generating that code now. Because the generated
1941 // code is a viable point for a memory access error, we make the pragmatic
1942 // choice here to directly use CI's location so that we have useful
1943 // attribution for the generated code.
1944 B.SetCurrentDebugLocation(CI->getDebugLoc());
1945
1946 BasicBlock *BBCI = CI->getParent();
1947 BasicBlock *BBTail =
1948 SplitBlock(BBCI, CI, DTU, nullptr, nullptr, BBCI->getName() + ".tail");
1949
1950 SmallVector<BasicBlock *> BBSubs;
1951 for (uint64_t I = 0; I < N; ++I)
1952 BBSubs.push_back(
1953 BasicBlock::Create(Ctx, "sub_" + Twine(I), BBCI->getParent(), BBTail));
1954 BasicBlock *BBNE = BasicBlock::Create(Ctx, "ne", BBCI->getParent(), BBTail);
1955
1956 cast<UncondBrInst>(BBCI->getTerminator())->setSuccessor(BBSubs[0]);
1957
1958 B.SetInsertPoint(BBNE);
1959 PHINode *Phi = B.CreatePHI(CI->getType(), N);
1960 B.CreateBr(BBTail);
1961
1962 Value *Base = LHS;
1963 for (uint64_t i = 0; i < N; ++i) {
1964 B.SetInsertPoint(BBSubs[i]);
1965 Value *VL =
1966 B.CreateZExt(B.CreateLoad(B.getInt8Ty(),
1967 B.CreateInBoundsPtrAdd(Base, B.getInt64(i))),
1968 CI->getType());
1969 Value *VR =
1970 ConstantInt::get(CI->getType(), static_cast<unsigned char>(RHS[i]));
1971 Value *Sub = Swapped ? B.CreateSub(VR, VL) : B.CreateSub(VL, VR);
1972 if (i < N - 1) {
1973 CondBrInst *CondBrInst = B.CreateCondBr(
1974 B.CreateICmpNE(Sub, ConstantInt::get(CI->getType(), 0)), BBNE,
1975 BBSubs[i + 1]);
1976
1977 Function *F = CI->getFunction();
1978 assert(F && "Instruction does not belong to a function!");
1979 std::optional<uint64_t> EC = F->getEntryCount();
1980 if (EC && *EC > 0)
1981 setExplicitlyUnknownBranchWeights(*CondBrInst, DEBUG_TYPE);
1982 } else {
1983 B.CreateBr(BBNE);
1984 }
1985
1986 Phi->addIncoming(Sub, BBSubs[i]);
1987 }
1988
1989 CI->replaceAllUsesWith(Phi);
1990 CI->eraseFromParent();
1991
1992 if (DTU) {
1993 SmallVector<DominatorTree::UpdateType, 8> Updates;
1994 Updates.push_back({DominatorTree::Insert, BBCI, BBSubs[0]});
1995 for (uint64_t i = 0; i < N; ++i) {
1996 if (i < N - 1)
1997 Updates.push_back({DominatorTree::Insert, BBSubs[i], BBSubs[i + 1]});
1998 Updates.push_back({DominatorTree::Insert, BBSubs[i], BBNE});
1999 }
2000 Updates.push_back({DominatorTree::Insert, BBNE, BBTail});
2001 Updates.push_back({DominatorTree::Delete, BBCI, BBTail});
2002 DTU->applyUpdates(Updates);
2003 }
2004}

2006/// Convert memchr with a small constant string into a switch
2007static bool foldMemChr(CallInst *Call, DomTreeUpdater *DTU,
2008 const DataLayout &DL) {
2009 if (isa<Constant>(Call->getArgOperand(1)))
2010 return false;
2011
2012 StringRef Str;
2013 Value *Base = Call->getArgOperand(0);
2014 if (!getConstantStringInfo(Base, Str, /*TrimAtNul=*/false))
2015 return false;
2016
2017 uint64_t N = Str.size();
2018 if (auto *ConstInt = dyn_cast<ConstantInt>(Call->getArgOperand(2))) {
2019 uint64_t Val = ConstInt->getZExtValue();
2020 // Ignore the case that n is larger than the size of string.
2021 if (Val > N)
2022 return false;
2023 N = Val;
2024 } else
2025 return false;
2026
2027 if (N > MemChrInlineThreshold)
2028 return false;
2029
2030 BasicBlock *BB = Call->getParent();
2031 BasicBlock *BBNext = SplitBlock(BB, Call, DTU);
2032 IRBuilder<> IRB(BB);
2033 IRB.SetCurrentDebugLocation(Call->getDebugLoc());
2034 IntegerType *ByteTy = IRB.getInt8Ty();
2035 BB->getTerminator()->eraseFromParent();
2036 SwitchInst *SI = IRB.CreateSwitch(
2037 IRB.CreateTrunc(Call->getArgOperand(1), ByteTy), BBNext, N);
2038 // We can't know the precise weights here, as they would depend on the value
2039 // distribution of Call->getArgOperand(1). So we just mark it as "unknown".
2040 setExplicitlyUnknownBranchWeightsIfProfiled(*SI, DEBUG_TYPE);
2041 Type *IndexTy = DL.getIndexType(Call->getType());
2042 SmallVector<DominatorTree::UpdateType, 8> Updates;
2043
2044 BasicBlock *BBSuccess = BasicBlock::Create(
2045 Call->getContext(), "memchr.success", BB->getParent(), BBNext);
2046 IRB.SetInsertPoint(BBSuccess);
2047 PHINode *IndexPHI = IRB.CreatePHI(IndexTy, N, "memchr.idx");
2048 Value *FirstOccursLocation = IRB.CreateInBoundsPtrAdd(Base, IndexPHI);
2049 IRB.CreateBr(BBNext);
2050 if (DTU)
2051 Updates.push_back({DominatorTree::Insert, BBSuccess, BBNext});
2052
2053 SmallPtrSet<ConstantInt *, 4> Cases;
2054 for (uint64_t I = 0; I < N; ++I) {
2055 ConstantInt *CaseVal =
2056 ConstantInt::get(ByteTy, static_cast<unsigned char>(Str[I]));
2057 if (!Cases.insert(CaseVal).second)
2058 continue;
2059
2060 BasicBlock *BBCase = BasicBlock::Create(Call->getContext(), "memchr.case",
2061 BB->getParent(), BBSuccess);
2062 SI->addCase(CaseVal, BBCase);
2063 IRB.SetInsertPoint(BBCase);
2064 IndexPHI->addIncoming(ConstantInt::get(IndexTy, I), BBCase);
2065 IRB.CreateBr(BBSuccess);
2066 if (DTU) {
2067 Updates.push_back({DominatorTree::Insert, BB, BBCase});
2068 Updates.push_back({DominatorTree::Insert, BBCase, BBSuccess});
2069 }
2070 }
2071
2072 PHINode *PHI =
2073 PHINode::Create(Call->getType(), 2, Call->getName(), BBNext->begin());
2074 PHI->addIncoming(Constant::getNullValue(Call->getType()), BB);
2075 PHI->addIncoming(FirstOccursLocation, BBSuccess);
2076
2077 Call->replaceAllUsesWith(PHI);
2078 Call->eraseFromParent();
2079
2080 if (DTU)
2081 DTU->applyUpdates(Updates);
2082
2083 return true;
2084}

2129/// Match high part of long multiplication.
2130///
2131/// Considering a multiply made up of high and low parts, we can split the
2132/// multiply into:
2133/// x * y == (xh*T + xl) * (yh*T + yl)
2134/// where xh == x>>32 and xl == x & 0xffffffff. T = 2^32.
2135/// This expands to
2136/// xh*yh*T*T + xh*yl*T + xl*yh*T + xl*yl
2137/// which can be drawn as
2138/// [ xh*yh ]
2139/// [ xh*yl ]
2140/// [ xl*yh ]
2141/// [ xl*yl ]
2142/// We are looking for the "high" half, which is xh*yh + xh*yl>>32 + xl*yh>>32 +
2143/// some carrys. The carry makes this difficult and there are multiple ways of
2144/// representing it. The ones we attempt to support here are:
2145/// Carry: xh*yh + carry + lowsum
2146/// carry = lowsum < xh*yl ? 0x1000000 : 0
2147/// lowsum = xh*yl + xl*yh + (xl*yl>>32)
2148/// Ladder: xh*yh + c2>>32 + c3>>32
2149/// c2 = xh*yl + (xl*yl>>32); c3 = c2&0xffffffff + xl*yh
2150/// or c2 = (xl*yh&0xffffffff) + xh*yl + (xl*yl>>32); c3 = xl*yh
2151/// Carry4: xh*yh + carry + crosssum>>32 + (xl*yl + crosssum&0xffffffff) >> 32
2152/// crosssum = xh*yl + xl*yh
2153/// carry = crosssum < xh*yl ? 0x1000000 : 0
2154/// Ladder4: xh*yh + (xl*yh)>>32 + (xh*yl)>>32 + low>>32;
2155/// low = (xl*yl)>>32 + (xl*yh)&0xffffffff + (xh*yl)&0xffffffff
2156///
2157/// They all start by matching xh*yh + 2 or 3 other operands. The bottom of the
2158/// tree is xh*yh, xh*yl, xl*yh and xl*yl.
2159static bool foldMulHigh(Instruction &I) {
2160 Type *Ty = I.getType();
2161 if (!Ty->isIntOrIntVectorTy())
2162 return false;
2163
2164 unsigned BitWidth = Ty->getScalarSizeInBits();
2165 APInt LowMask = APInt::getLowBitsSet(BitWidth, BitWidth / 2);
2166 if (BitWidth % 2 != 0)
2167 return false;
2168
2169 auto CreateMulHigh = [&](Value *X, Value *Y) {
2170 IRBuilder<> Builder(&I);
2171 Type *NTy = Ty->getWithNewBitWidth(BitWidth * 2);
2172 Value *XExt = Builder.CreateZExt(X, NTy);
2173 Value *YExt = Builder.CreateZExt(Y, NTy);
2174 Value *Mul = Builder.CreateMul(XExt, YExt, "", /*HasNUW=*/true);
2175 Value *High = Builder.CreateLShr(Mul, BitWidth);
2176 Value *Res = Builder.CreateTrunc(High, Ty, "", /*HasNUW=*/true);
2177 Res->takeName(&I);
2178 I.replaceAllUsesWith(Res);
2179 LLVM_DEBUG(dbgs() << "Created long multiply from parts of " << *X << " and "
2180 << *Y << "\n");
2181 return true;
2182 };

2184 // Common check routines for X_lo*Y_lo and X_hi*Y_lo
2185 auto CheckLoLo = [&](Value *XlYl, Value *X, Value *Y) {
2186 return match(XlYl, m_c_Mul(m_And(m_Specific(X), m_SpecificInt(LowMask)),
2187 m_And(m_Specific(Y), m_SpecificInt(LowMask))));
2188 };
2189 auto CheckHiLo = [&](Value *XhYl, Value *X, Value *Y) {
2190 return match(XhYl,
2191 m_c_Mul(m_LShr(m_Specific(X), m_SpecificInt(BitWidth / 2)),
2192 m_And(m_Specific(Y), m_SpecificInt(LowMask))));
2193 };
2194
2195 auto FoldMulHighCarry = [&](Value *X, Value *Y, Instruction *Carry,
2196 Instruction *B) {
2197 // Looking for LowSum >> 32 and carry (select)
2198 if (Carry->getOpcode() != Instruction::Select)
2199 std::swap(Carry, B);
2200
2201 // Carry = LowSum < XhYl ? 0x100000000 : 0
2202 Value *LowSum, *XhYl;
2203 if (!match(Carry,
2204 m_OneUse(m_Select(
2205 m_OneUse(m_SpecificICmp(ICmpInst::ICMP_ULT, m_Value(LowSum),
2206 m_Value(XhYl))),
2207 m_SpecificInt(APInt::getOneBitSet(BitWidth, BitWidth / 2)),
2208 m_Zero()))))
2209 return false;
2210
2211 // XhYl can be Xh*Yl or Xl*Yh
2212 if (!CheckHiLo(XhYl, X, Y)) {
2213 if (CheckHiLo(XhYl, Y, X))
2214 std::swap(X, Y);
2215 else
2216 return false;
2217 }
2218 if (XhYl->hasNUsesOrMore(3))
2219 return false;
2220
2221 // B = LowSum >> 32
2222 if (!match(B, m_OneUse(m_LShr(m_Specific(LowSum),
2223 m_SpecificInt(BitWidth / 2)))) ||
2224 LowSum->hasNUsesOrMore(3))
2225 return false;
2226
2227 // LowSum = XhYl + XlYh + XlYl>>32
2228 Value *XlYh, *XlYl;
2229 auto XlYlHi = m_LShr(m_Value(XlYl), m_SpecificInt(BitWidth / 2));
2230 if (!match(LowSum,
2231 m_c_Add(m_Specific(XhYl),
2232 m_OneUse(m_c_Add(m_OneUse(m_Value(XlYh)), XlYlHi)))) &&
2233 !match(LowSum, m_c_Add(m_OneUse(m_Value(XlYh)),
2234 m_OneUse(m_c_Add(m_Specific(XhYl), XlYlHi)))) &&
