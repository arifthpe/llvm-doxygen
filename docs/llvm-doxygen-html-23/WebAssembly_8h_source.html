115enum TargetIndex {
116 // Followed by a local index (ULEB).
117 TI_LOCAL,
118 // Followed by an absolute global index (ULEB). DEPRECATED.
119 TI_GLOBAL_FIXED,
120 // Followed by the index from the bottom of the Wasm stack.
121 TI_OPERAND_STACK,
122 // Followed by a compilation unit relative global index (uint32_t)
123 // that will have an associated relocation.
124 TI_GLOBAL_RELOC,
125 // Like TI_LOCAL, but indicates an indirect value (e.g. byval arg
126 // passed by pointer).
127 TI_LOCAL_INDIRECT
