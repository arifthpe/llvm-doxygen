64static void emitEpiloguePACSymOffsetIntoReg(const TargetInstrInfo &TII,
65 MachineBasicBlock &MBB,
66 MachineBasicBlock::iterator I,
67 DebugLoc DL, MCSymbol *PACSym,
68 Register Reg) {
69 BuildMI(MBB, I, DL, TII.get(AArch64::ADRP), Reg)
70 .addSym(PACSym, AArch64II::MO_PAGE)
71 .setMIFlag(MachineInstr::FrameDestroy);
72 BuildMI(MBB, I, DL, TII.get(AArch64::ADDXri), Reg)
73 .addReg(Reg)
74 .addSym(PACSym, AArch64II::MO_PAGEOFF | AArch64II::MO_NC)
75 .addImm(0)
76 .setMIFlag(MachineInstr::FrameDestroy);
77}

79static void emitPACCFI(MachineBasicBlock &MBB, MachineBasicBlock::iterator MBBI,
80 MachineInstr::MIFlag Flags, bool EmitCFI) {
81 if (!EmitCFI)
82 return;
83
84 auto &MF = *MBB.getParent();
85 auto &MFnI = *MF.getInfo<AArch64FunctionInfo>();
86
87 // DW_CFA_AARCH64_negate_ra_state_with_pc is semantically broken for
88 // functions where shrinkwrapping places signing/authenticating pairs on
89 // distinct CFG paths.
90 //
91 // DWARF CFI is evaluated linearly over the byte stream, not along control
92 // flow edges. The toggle semantics of this directive therefore cannot
93 // faithfully represent the signed/unsigned RA state for all possible CFG
94 // paths. The added complexity versus DW_CFA_AARCH64_negate_ra_state is that
95 // an unwinder must also reconstruct the PC of the PACI[AB]SPPC in order to
96 // verify the signed LR, and that address is derived from the location of this
97 // directive in the linear CFI stream.
98 //
99 // The correct fix is to use DW_CFA_AARCH64_set_ra_state_with_pc, which sets
100 // the RA state and signing address absolutely rather than toggling them. An
101 // unwinder that supports this directive can reconstruct the correct state on
102 // any CFG path, regardless of how many signing/authenticating pairs exist in
103 // the function. However, not all unwinders support this directive, so we
104 // cannot rely on it exclusively.
105 //
106 // For unwinders that only support DW_CFA_AARCH64_negate_ra_state_with_pc,
107 // libunwind exploits a loophole: it records the address at the
108 // DW_CFA_AARCH64_negate_ra_state_with_pc site to authenticate the LR, but
109 // does not care that the CFI state remains "signed with pc" after
110 // authentication has occurred. This means we can safely omit the
111 // FrameDestroy emission of this directive, treating it solely as a marker
112 // for the signing site, as long as each function has at most one such
113 // signing location. That invariant holds today because shrinkwrapping
114 // does not yet hoist or sink PAuth_LR frame code across CFG join/split
115 // points; once it does, we must avoid those transformations on platforms that
116 // have this limitation.
117 //
118 // https://github.com/ARM-software/abi-aa/issues/327
119 // https://github.com/ARM-software/abi-aa/pull/346
120 if (Flags == MachineInstr::FrameDestroy && MFnI.branchProtectionPAuthLR())
121 return;
122
123 CFIInstBuilder CFIBuilder(MBB, MBBI, Flags);
124 if (MFnI.branchProtectionPAuthLR()) {
125 CFIBuilder.buildNegateRAStateWithPC();
126 } else if (!MF.getTarget().getTargetTriple().isOSBinFormatMachO()) {
127 CFIBuilder.buildNegateRAState();
128 }
129}

131void AArch64PointerAuthImpl::signLR(MachineFunction &MF,
132 MachineBasicBlock::iterator MBBI) const {
133 auto &MFnI = *MF.getInfo<AArch64FunctionInfo>();
134 bool UseBKey = MFnI.shouldSignWithBKey();
135 bool EmitCFI = MFnI.needsDwarfUnwindInfo(MF);
136 bool NeedsWinCFI = MF.hasWinCFI();
137
138 MachineBasicBlock &MBB = *MBBI->getParent();
139
140 // Debug location must be unknown, see AArch64FrameLowering::emitPrologue.
141 DebugLoc DL;
142
143 if (UseBKey && !MF.getTarget().getTargetTriple().isOSBinFormatMachO()) {
144 BuildMI(MBB, MBBI, DL, TII->get(AArch64::EMITBKEY))
145 .setMIFlag(MachineInstr::FrameSetup);
146 }
147
148 // PAuthLR authentication instructions need to know the value of PC at the
149 // point of signing (PACI*).
150 if (MFnI.branchProtectionPAuthLR()) {
151 MCSymbol *PACSym = MF.getContext().createTempSymbol();
152 MFnI.setSigningInstrLabel(PACSym);
153 }
154
155 // No SEH opcode for this one; it doesn't materialize into an
156 // instruction on Windows.
157 if (MFnI.branchProtectionPAuthLR() && Subtarget->hasPAuthLR()) {
158 emitPACCFI(MBB, MBBI, MachineInstr::FrameSetup, EmitCFI);
159 BuildMI(MBB, MBBI, DL,
160 TII->get(UseBKey ? AArch64::PACIBSPPC : AArch64::PACIASPPC))
161 .setMIFlag(MachineInstr::FrameSetup)
162 ->setPreInstrSymbol(MF, MFnI.getSigningInstrLabel());
163 } else {
164 if (MFnI.branchProtectionPAuthLR()) {
165 BuildMI(MBB, MBBI, DL, TII->get(AArch64::PACM))
166 .setMIFlag(MachineInstr::FrameSetup);
167 emitPACCFI(MBB, MBBI, MachineInstr::FrameSetup, EmitCFI);
168 }
169 BuildMI(MBB, MBBI, DL,
170 TII->get(UseBKey ? AArch64::PACIBSP : AArch64::PACIASP))
171 .setMIFlag(MachineInstr::FrameSetup)
172 ->setPreInstrSymbol(MF, MFnI.getSigningInstrLabel());
173 if (!MFnI.branchProtectionPAuthLR())
174 emitPACCFI(MBB, MBBI, MachineInstr::FrameSetup, EmitCFI);
175 }
176
177 if (!EmitCFI && NeedsWinCFI) {
178 assert(UseBKey &&
179 "Windows SEH PAC unwind info only supports B-key signing");
180 BuildMI(MBB, MBBI, DL, TII->get(AArch64::SEH_PACSignLR))
181 .setMIFlag(MachineInstr::FrameSetup);
182 }
183}

185void AArch64PointerAuthImpl::authenticateLR(
186 MachineFunction &MF, MachineBasicBlock::iterator MBBI) const {
187 const AArch64FunctionInfo *MFnI = MF.getInfo<AArch64FunctionInfo>();
188 bool UseBKey = MFnI->shouldSignWithBKey();
189 bool EmitAsyncCFI = MFnI->needsAsyncDwarfUnwindInfo(MF);
190 bool NeedsWinCFI = MF.hasWinCFI();
191
192 MachineBasicBlock &MBB = *MBBI->getParent();
193 DebugLoc DL = MBBI->getDebugLoc();
194 // MBBI points to a PAUTH_EPILOGUE instruction to be replaced and
195 // TI points to a terminator instruction that may or may not be combined.
196 // Note that inserting new instructions "before MBBI" and "before TI" is
197 // not the same because if ShadowCallStack is enabled, its instructions
198 // are placed between MBBI and TI.
199 MachineBasicBlock::iterator TI = MBB.getFirstInstrTerminator();
200
201 // The AUTIASP instruction assembles to a hint instruction before v8.3a so
202 // this instruction can safely used for any v8a architecture.
203 // From v8.3a onwards there are optimised authenticate LR and return
204 // instructions, namely RETA{A,B}, that can be used instead. In this case the
205 // DW_CFA_AARCH64_negate_ra_state can't be emitted.
206 bool TerminatorIsCombinable =
207 TI != MBB.end() && TI->getOpcode() == AArch64::RET;
208 MCSymbol *PACSym = MFnI->getSigningInstrLabel();
209
210 if (Subtarget->hasPAuth() && TerminatorIsCombinable && !NeedsWinCFI &&
211 !MF.getFunction().hasFnAttribute(Attribute::ShadowCallStack)) {
212 if (MFnI->branchProtectionPAuthLR() && Subtarget->hasPAuthLR()) {
213 assert(PACSym && "No PAC instruction to refer to");
214 BuildMI(MBB, TI, DL,
215 TII->get(UseBKey ? AArch64::RETABSPPCi : AArch64::RETAASPPCi))
216 .addSym(PACSym)
217 .copyImplicitOps(*MBBI)
218 .setMIFlag(MachineInstr::FrameDestroy);
219 } else {
220 if (MFnI->branchProtectionPAuthLR()) {
221 emitEpiloguePACSymOffsetIntoReg(*TII, MBB, MBBI, DL, PACSym,
222 AArch64::X16);
223 BuildMI(MBB, MBBI, DL, TII->get(AArch64::PACM))
224 .setMIFlag(MachineInstr::FrameDestroy);
225 }
226 BuildMI(MBB, TI, DL, TII->get(UseBKey ? AArch64::RETAB : AArch64::RETAA))
227 .copyImplicitOps(*MBBI)
228 .setMIFlag(MachineInstr::FrameDestroy);
229 }
230 MBB.erase(TI);
231 return;
232 }
233
234 auto &AFL = *static_cast<const AArch64FrameLowering *>(
235 MF.getSubtarget().getFrameLowering());
236 int64_t ArgumentStackToRestore = AFL.getArgumentStackToRestore(MF, MBB);
237
238 // When ArgumentStackToRestore < 0, the tail callee pops more argument space
239 // than this function received, so after the frame teardown SP is below the
240 // entry SP used as the signing modifier. Reconstruct entry SP in x16 and
241 // authenticate using AUTI[AB]1716 (x17=LR, x16=entry_SP).
242 if (ArgumentStackToRestore < 0) {
243 emitFrameOffset(MBB, MBBI, DL, AArch64::X16, AArch64::SP,
244 StackOffset::getFixed(-ArgumentStackToRestore), TII,
245 MachineInstr::FrameDestroy);
246
247 BuildMI(MBB, MBBI, DL, TII->get(AArch64::ORRXrs), AArch64::X17)
248 .addReg(AArch64::XZR)
249 .addReg(AArch64::LR)
250 .addImm(0)
251 .setMIFlag(MachineInstr::FrameDestroy);
252
253 if (MFnI->branchProtectionPAuthLR() && Subtarget->hasPAuthLR()) {
254 assert(PACSym && "No PAC instruction to refer to");
255 emitEpiloguePACSymOffsetIntoReg(*TII, MBB, MBBI, DL, PACSym,
256 AArch64::X15);
257
258 emitPACCFI(MBB, MBBI, MachineInstr::FrameDestroy, EmitAsyncCFI);
259 unsigned AutOpc = UseBKey ? AArch64::AUTIB171615 : AArch64::AUTIA171615;
260 BuildMI(MBB, MBBI, DL, TII->get(AutOpc))
261 .setMIFlag(MachineInstr::FrameDestroy);
262 } else if (MFnI->branchProtectionPAuthLR()) {
263 assert(PACSym && "No PAC instruction to refer to");
264 emitEpiloguePACSymOffsetIntoReg(*TII, MBB, MBBI, DL, PACSym,
265 AArch64::X15);
266
267 // The PACM hint-space instruction modifies the following AUTI[AB]1716
268 // to optionally take x15 as an extra operand depending on the
269 // presence of +pauth-lr at runtime. On machines without +pauth-lr, it
270 // behaves as a nop, and the address of the PACI[AB]SP in x15 is
271 // ignored.
272 BuildMI(MBB, MBBI, DL, TII->get(AArch64::PACM))
273 .setMIFlag(MachineInstr::FrameDestroy);
274
275 emitPACCFI(MBB, MBBI, MachineInstr::FrameDestroy, EmitAsyncCFI);
276 unsigned AutOpc = UseBKey ? AArch64::AUTIB1716 : AArch64::AUTIA1716;
277 BuildMI(MBB, MBBI, DL, TII->get(AutOpc))
278 .setMIFlag(MachineInstr::FrameDestroy);
279 } else {
280 unsigned AutOpc = UseBKey ? AArch64::AUTIB1716 : AArch64::AUTIA1716;
281 BuildMI(MBB, MBBI, DL, TII->get(AutOpc))
282 .setMIFlag(MachineInstr::FrameDestroy);
283 emitPACCFI(MBB, MBBI, MachineInstr::FrameDestroy, EmitAsyncCFI);
284 }
285
286 BuildMI(MBB, MBBI, DL, TII->get(AArch64::ORRXrs), AArch64::LR)
287 .addReg(AArch64::XZR)
288 .addReg(AArch64::X17)
289 .addImm(0)
290 .setMIFlag(MachineInstr::FrameDestroy);
291 return;
292 }
293
294 // When ArgumentStackToRestore > 0, this function received more argument
295 // space than the tail callee pops. The epilogue contains an SP adjustment
296 // (e.g. "add sp, sp, #N") to discard the leftover argument space. We must
297 // authenticate *before* that adjustment so that AUTI[AB]SP sees the entry
298 // SP discriminator. Move any such SP-adjusting instructions to after the
299 // authentication instruction.
300 //
301 // We cannot simply bump SP first and then use AUTI[AB]SP with the bumped
302 // value, because the live arguments would fall below SP and potentially
303 // outside the red-zone.
304 SmallVector<MachineInstr *, 2> SPMods;
305 if (ArgumentStackToRestore > 0) {
306 for (auto I = MBBI; I->getFlag(MachineInstr::FrameDestroy); --I) {
307 if ((I->getOpcode() == AArch64::ADDXri ||
308 I->getOpcode() == AArch64::SUBXri) &&
309 I->getOperand(0).getReg() == AArch64::SP &&
310 I->getOperand(1).getReg() == AArch64::SP)
311 SPMods.push_back(&*I);
312 }
313 }
314 for (auto *MI : SPMods)
315 MI->removeFromParent();
316
317 if (MFnI->branchProtectionPAuthLR() && Subtarget->hasPAuthLR()) {
318 assert(PACSym && "No PAC instruction to refer to");
319 emitPACCFI(MBB, MBBI, MachineInstr::FrameDestroy, EmitAsyncCFI);
320 BuildMI(MBB, MBBI, DL,
321 TII->get(UseBKey ? AArch64::AUTIBSPPCi : AArch64::AUTIASPPCi))
322 .addSym(PACSym)
323 .setMIFlag(MachineInstr::FrameDestroy);
324 } else {
325 if (MFnI->branchProtectionPAuthLR()) {
326 emitEpiloguePACSymOffsetIntoReg(*TII, MBB, MBBI, DL, PACSym,
327 AArch64::X16);
328
329 BuildMI(MBB, MBBI, DL, TII->get(AArch64::PACM))
330 .setMIFlag(MachineInstr::FrameDestroy);
331 emitPACCFI(MBB, MBBI, MachineInstr::FrameDestroy, EmitAsyncCFI);
332 }
333 BuildMI(MBB, MBBI, DL,
334 TII->get(UseBKey ? AArch64::AUTIBSP : AArch64::AUTIASP))
335 .setMIFlag(MachineInstr::FrameDestroy);
336 if (!MFnI->branchProtectionPAuthLR())
337 emitPACCFI(MBB, MBBI, MachineInstr::FrameDestroy, EmitAsyncCFI);
338 }
339
340 if (NeedsWinCFI) {
341 assert(UseBKey &&
342 "Windows SEH PAC unwind info only supports B-key signing");
343 BuildMI(MBB, MBBI, DL, TII->get(AArch64::SEH_PACSignLR))
344 .setMIFlag(MachineInstr::FrameDestroy);
345 }
346
347 for (auto *MI : SPMods)
348 MBB.insert(MBBI, MI);
349}

351unsigned llvm::AArch64PAuth::getCheckerSizeInBytes(AuthCheckMethod Method) {
352 switch (Method) {
353 case AuthCheckMethod::None:
354 return 0;
355 case AuthCheckMethod::DummyLoad:
356 return 4;
357 case AuthCheckMethod::HighBitsNoTBI:
358 return 12;
359 case AuthCheckMethod::XPACHint:
360 case AuthCheckMethod::XPAC:
361 return 20;
362 }
363 llvm_unreachable("Unknown AuthCheckMethod enum");
364}

366bool AArch64PointerAuthImpl::run(MachineFunction &MF) {
367 Subtarget = &MF.getSubtarget<AArch64Subtarget>();
368 TII = Subtarget->getInstrInfo();
369
370 SmallVector<MachineBasicBlock::instr_iterator> PAuthPseudoInstrs;
371
372 bool Modified = false;
373
374 for (auto &MBB : MF) {
375 for (auto &MI : MBB) {
376 switch (MI.getOpcode()) {
377 default:
378 break;
379 case AArch64::PAUTH_PROLOGUE:
380 case AArch64::PAUTH_EPILOGUE:
381 PAuthPseudoInstrs.push_back(MI.getIterator());
382 break;
383 }
384 }
385 }
386
387 for (auto It : PAuthPseudoInstrs) {
388 switch (It->getOpcode()) {
389 case AArch64::PAUTH_PROLOGUE:
390 signLR(MF, It);
391 break;
392 case AArch64::PAUTH_EPILOGUE:
393 authenticateLR(MF, It);
394 break;
395 default:
396 llvm_unreachable("Unhandled opcode");
397 }
398 It->eraseFromParent();
399 Modified = true;
400 }
401
402 return Modified;
403}
