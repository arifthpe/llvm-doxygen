79 template <size_t IndexSize>
80 void printPCIBDMem(const InstTy *MI, uint64_t Address, unsigned OpNum,
81 raw_ostream &O) {
82 static_assert(IndexSize == 16 || IndexSize == 32, "unsupported index size");
83
84 O << '(';
85 impl().printDisp(MI, OpNum + M68k::PCRelDisp, O);
86 O << ",%pc";
87 std::string RegStr;
88 raw_string_ostream SS(RegStr);
89 impl().printOperand(MI, OpNum + M68k::PCRelIndex, SS);
90 if (!RegStr.empty()) {
91 O << "," << RegStr;
92 O << (IndexSize == 16 ? ".W" : ".L");
93 }
94 impl().printScale(MI, OpNum + M68k::PCRelScale, O);
95 O << ')';
96 }
