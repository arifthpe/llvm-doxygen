190 /// Remove the last element from the vector.
191 void pop_back() {
192 if constexpr (canBeSmall())
193 if (isSmall()) {

203 /// Remove the element given by Iterator.
204 ///
205 /// Returns an iterator to the element following the one which was removed,
206 /// which may be end().
207 ///
208 /// \note This is a deceivingly expensive operation (linear time). It's
209 /// usually better to use \a remove_if() if possible.
210 typename VectorType::iterator erase(typename VectorType::iterator Iterator) {
211 if constexpr (canBeSmall())
212 if (isSmall())
213 return Vector.erase(Iterator);
214
215 Map.erase(Iterator->first);
216 auto Next = Vector.erase(Iterator);
217 if (Next == Vector.end())
218 return Next;
219
220 // Update indices in the map.
221 size_t Index = Next - Vector.begin();
222 for (auto &I : Map) {
223 assert(I.second != Index && "Index was already erased!");
224 if (I.second > Index)
225 --I.second;
226 }
227 return Next;
228 }

230 /// Remove all elements with the key value Key.
231 ///
232 /// Returns the number of elements removed.
233 size_type erase(const KeyT &Key) {
234 auto Iterator = find(Key);
235 if (Iterator == end())
236 return 0;
237 erase(Iterator);
238 return 1;
239 }

282 template <typename KeyArgT, typename... Ts>
283 std::pair<iterator, bool> try_emplace_impl(KeyArgT &&Key, Ts &&...Args) {
284 if constexpr (canBeSmall())
285 if (isSmall()) {
286 auto I = findInVector(Vector, Key);
287 if (I != Vector.end())
288 return {I, false};
289 Vector.emplace_back(std::piecewise_construct,
290 std::forward_as_tuple(std::forward<KeyArgT>(Key)),
291 std::forward_as_tuple(std::forward<Ts>(Args)...));
292 if (Vector.size() > N)
293 makeBig();
294 return {std::prev(end()), true};
295 }
296
297 auto [It, Inserted] = Map.try_emplace(Key);
298 if (Inserted) {
299 It->second = Vector.size();
300 Vector.emplace_back(std::piecewise_construct,
301 std::forward_as_tuple(std::forward<KeyArgT>(Key)),
302 std::forward_as_tuple(std::forward<Ts>(Args)...));
303 return {std::prev(end()), true};
304 }
305 return {begin() + It->second, false};
306 }

309template <typename KeyT, typename ValueT, typename MapType, typename VectorType,
310 unsigned N>
311template <class Function>
312void MapVector<KeyT, ValueT, MapType, VectorType, N>::remove_if(Function Pred) {
313 if constexpr (canBeSmall())
314 if (isSmall()) {
315 Vector.erase(llvm::remove_if(Vector, Pred), Vector.end());
316 return;
317 }
318
319 auto O = Vector.begin();
320 for (auto I = O, E = Vector.end(); I != E; ++I) {
321 if (Pred(*I)) {
322 // Erase from the map.
323 Map.erase(I->first);
324 continue;
325 }
326
327 if (I != O) {
328 // Move the value and update the index in the map.
329 *O = std::move(*I);
330 Map[O->first] = O - Vector.begin();
331 }
332 ++O;
333 }
334 // Erase trailing entries in the vector.
335 Vector.erase(O, Vector.end());
336}
