75// Force static initialization.
76extern "C" LLVM_ABI LLVM_EXTERNAL_VISIBILITY void
77LLVMInitializeNVPTXTargetMC() {
78 for (Target *T : {&getTheNVPTXTarget32(), &getTheNVPTXTarget64()}) {
79 // Register the MC asm info.
80 RegisterMCAsmInfo<NVPTXMCAsmInfo> X(*T);
81
82 // Register the MC instruction info.
83 TargetRegistry::RegisterMCInstrInfo(*T, createNVPTXMCInstrInfo);
84
85 // Register the MC register info.
86 TargetRegistry::RegisterMCRegInfo(*T, createNVPTXMCRegisterInfo);
87
88 // Register the MC subtarget info.
89 TargetRegistry::RegisterMCSubtargetInfo(*T, createNVPTXMCSubtargetInfo);
90
91 // Register the MCInstPrinter.
92 TargetRegistry::RegisterMCInstPrinter(*T, createNVPTXMCInstPrinter);
93
94 // Register the MCTargetStreamer.
95 TargetRegistry::RegisterAsmTargetStreamer(*T, createTargetAsmStreamer);
96
97 // Register the MCTargetStreamer.
98 TargetRegistry::RegisterNullTargetStreamer(*T, createNullTargetStreamer);
99 }
100}
