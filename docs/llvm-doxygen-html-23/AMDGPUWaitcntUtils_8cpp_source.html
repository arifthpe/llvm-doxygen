51HardwareLimits::HardwareLimits(const IsaVersion &IV) {
52 bool HasExtendedWaitCounts = IV.Major >= 12;
53 if (HasExtendedWaitCounts) {
54 LoadcntMax = getLoadcntBitMask(IV);
55 DscntMax = getDscntBitMask(IV);
56 } else {
57 LoadcntMax = getVmcntBitMask(IV);
58 DscntMax = getLgkmcntBitMask(IV);
59 }
60 ExpcntMax = getExpcntBitMask(IV);
61 StorecntMax = getStorecntBitMask(IV);
62 SamplecntMax = getSamplecntBitMask(IV);
63 BvhcntMax = getBvhcntBitMask(IV);
64 KmcntMax = getKmcntBitMask(IV);
65 XcntMax = getXcntBitMask(IV);
66 AsyncMax = getAsynccntBitMask(IV);
67 VaVdstMax = DepCtr::getVaVdstBitMask();
68 VmVsrcMax = DepCtr::getVmVsrcBitMask();
69}

102Waitcnt decodeWaitcnt(const IsaVersion &Version, unsigned Encoded) {
103 Waitcnt Decoded;
104 Decoded.set(LOAD_CNT, decodeVmcnt(Version, Encoded));
105 Decoded.set(EXP_CNT, decodeExpcnt(Version, Encoded));
106 Decoded.set(DS_CNT, decodeLgkmcnt(Version, Encoded));
107 return Decoded;
