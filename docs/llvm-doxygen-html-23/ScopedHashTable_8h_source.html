9// This file implements an efficient scoped hash table, which is useful for
10// things like dominator-based optimizations. This allows clients to do things
11// like this:
12//
13// ScopedHashTable<int, int> HT;
14// {
15// ScopedHashTableScope<int, int> Scope1(HT);
16// HT.insert(0, 0);
17// HT.insert(1, 1);
18// {
19// ScopedHashTableScope<int, int> Scope2(HT);
20// HT.insert(0, 42);
21// }
22// }
23//
24// Looking up the value for "0" in the Scope2 block will return 42. Looking
25// up the value for 0 before 42 is inserted or after Scope2 is popped will
26// return 0.
27//
28//===----------------------------------------------------------------------===//

45template <typename K, typename V>
46class ScopedHashTableVal {
47 ScopedHashTableVal *NextInScope;
48 ScopedHashTableVal *NextForKey;
49 ScopedHashTableVal *PreInScope;
50 K Key;
51 V Val;
52
53 ScopedHashTableVal(const K &key, const V &val) : Key(key), Val(val) {}
54
55public:
56 const K &getKey() const { return Key; }
57 const V &getValue() const { return Val; }
58 V &getValue() { return Val; }
59
60 ScopedHashTableVal *getNextForKey() { return NextForKey; }
61 const ScopedHashTableVal *getNextForKey() const { return NextForKey; }
62 ScopedHashTableVal *getNextInScope() { return NextInScope; }
63 ScopedHashTableVal *getPreInScope() { return PreInScope; }

65 template <typename AllocatorTy>
66 static ScopedHashTableVal *Create(ScopedHashTableVal *nextInScope,
67 ScopedHashTableVal *nextForKey,
68 const K &key, const V &val,
69 AllocatorTy &Allocator) {
70 ScopedHashTableVal *New = Allocator.template Allocate<ScopedHashTableVal>();
71 // Set up the value.
72 new (New) ScopedHashTableVal(key, val);
73 New->NextInScope = nextInScope;
74 New->NextForKey = nextForKey;
75 New->PreInScope = nullptr;
76 if (nextInScope)
77 nextInScope->PreInScope = New;
78 return New;
79 }

81 template <typename AllocatorTy> void Destroy(AllocatorTy &Allocator) {
82 // Free memory referenced by the item.
83 this->~ScopedHashTableVal();
84 Allocator.Deallocate(this);
85 }
86
87 template <typename AllocatorTy>
88 static void erase(ScopedHashTableVal<K, V> *&ThisEntry,
89 AllocatorTy &Allocator) {
90 ScopedHashTableVal<K, V> *ToDestroy = ThisEntry;
91 ScopedHashTableVal<K, V> *NextInScope = ThisEntry->NextInScope;
92 ScopedHashTableVal<K, V> *PrevInScope = ThisEntry->PreInScope;
93 if (PrevInScope)
94 PrevInScope->NextInScope = NextInScope;
95 if (NextInScope)
96 NextInScope->PreInScope = PrevInScope;
97 ThisEntry = ThisEntry->NextForKey;
98 ToDestroy->Destroy(Allocator);
99 }
100};

102template <typename K, typename V, typename KInfo = DenseMapInfo<K>,
103 typename AllocatorTy = MallocAllocator>
104class ScopedHashTableScope {
105 /// HT - The hashtable that we are active for.
106 ScopedHashTable<K, V, KInfo, AllocatorTy> &HT;
107
108 /// PrevScope - This is the scope that we are shadowing in HT.
109 ScopedHashTableScope *PrevScope;
110
111 /// LastValInScope - This is the last value that was inserted for this scope

124
125private:

128 ScopedHashTableVal<K, V> *getLastValInScope() {

137template <typename K, typename V, typename KInfo = DenseMapInfo<K>>
138class ScopedHashTableIterator {
139 ScopedHashTableVal<K, V> *Node;
140
141public:
142 ScopedHashTableIterator(ScopedHashTableVal<K, V> *node) : Node(node) {}
143
144 V &operator*() const {
145 assert(Node && "Dereference end()");
146 return Node->getValue();
147 }
148 V *operator->() const {
149 return &Node->getValue();
150 }
151
152 bool operator==(const ScopedHashTableIterator &RHS) const {
153 return Node == RHS.Node;
154 }
155 bool operator!=(const ScopedHashTableIterator &RHS) const {
156 return Node != RHS.Node;
157 }
158
159 inline ScopedHashTableIterator& operator++() { // Preincrement
160 assert(Node && "incrementing past end()");
161 Node = Node->getNextForKey();
162 return *this;
163 }
164 ScopedHashTableIterator operator++(int) { // Postincrement
165 ScopedHashTableIterator tmp = *this; ++*this; return tmp;
166 }
167};

169template <typename K, typename V, typename KInfo, typename AllocatorTy>
170class ScopedHashTable : detail::AllocatorHolder<AllocatorTy> {
171 using AllocTy = detail::AllocatorHolder<AllocatorTy>;
172
173public:
174 /// ScopeTy - A type alias for easy access to the name of the scope for this
175 /// hash table.
176 using ScopeTy = ScopedHashTableScope<K, V, KInfo, AllocatorTy>;
177 using size_type = unsigned;
178
179private:
180 friend class ScopedHashTableScope<K, V, KInfo, AllocatorTy>;
181
182 using ValTy = ScopedHashTableVal<K, V>;
183
184 DenseMap<K, ValTy*, KInfo> TopLevelMap;
185 ScopeTy *CurScope = nullptr;
186
187public:
188 ScopedHashTable() = default;
189 ScopedHashTable(AllocatorTy A) : AllocTy(A) {}
190 ScopedHashTable(const ScopedHashTable &) = delete;
191 ScopedHashTable &operator=(const ScopedHashTable &) = delete;
192
193 ~ScopedHashTable() {
194 assert(!CurScope && TopLevelMap.empty() && "Scope imbalance!");
195 }
196
197 /// Access to the allocator.
198 using AllocTy::getAllocator;
199
200 /// Return 1 if the specified key is in the table, 0 otherwise.
201 size_type count(const K &Key) const {
202 return TopLevelMap.count(Key);
203 }
204
205 V lookup(const K &Key) const {
206 auto I = TopLevelMap.find(Key);
207 if (I != TopLevelMap.end())
208 return I->second->getValue();
209
210 return V();
211 }
212
213 void insert(const K &Key, const V &Val) {
214 insertIntoScope(CurScope, Key, Val);
215 }
216
217 using iterator = ScopedHashTableIterator<K, V, KInfo>;
218
219 iterator end() { return iterator(nullptr); }
220
221 iterator begin(const K &Key) {
222 auto I = TopLevelMap.find(Key);
223 if (I == TopLevelMap.end()) return end();
224 return iterator(I->second);
225 }
226
227 ScopeTy *getCurScope() { return CurScope; }
228 const ScopeTy *getCurScope() const { return CurScope; }
229
230 /// insertIntoScope - This inserts the specified key/value at the specified
231 /// (possibly not the current) scope. While it is ok to insert into a scope
232 /// that isn't the current one, it isn't ok to insert *underneath* an existing
233 /// value of the specified key.
234 void insertIntoScope(ScopeTy *S, const K &Key, const V &Val) {
235 assert(S && "No scope active!");
236 ScopedHashTableVal<K, V> *&KeyEntry = TopLevelMap[Key];
237 KeyEntry = ValTy::Create(S->getLastValInScope(), KeyEntry, Key, Val,
238 getAllocator());
239 S->setLastValInScope(KeyEntry);
240 }
241
242 void erase(const K &key) { CurScope->erase(key); }
243};

245/// ScopedHashTableScope ctor - Install this as the current scope for the hash
246/// table.
247template <typename K, typename V, typename KInfo, typename Allocator>
248ScopedHashTableScope<K, V, KInfo, Allocator>::
249 ScopedHashTableScope(ScopedHashTable<K, V, KInfo, Allocator> &ht) : HT(ht) {
250 PrevScope = HT.CurScope;
251 HT.CurScope = this;
252 LastValInScope = nullptr;
253}
254
255template <typename K, typename V, typename KInfo, typename Allocator>
256ScopedHashTableScope<K, V, KInfo, Allocator>::~ScopedHashTableScope() {
257 assert(HT.CurScope == this && "Scope imbalance!");
258 HT.CurScope = PrevScope;
259
260 // Pop and delete all values corresponding to this scope.
261 while (ScopedHashTableVal<K, V> *ThisEntry = LastValInScope) {
262 // Pop this value out of the TopLevelMap.
263 if (!ThisEntry->getNextForKey()) {
264 assert(HT.TopLevelMap[ThisEntry->getKey()] == ThisEntry &&
265 "Scope imbalance!");
266 HT.TopLevelMap.erase(ThisEntry->getKey());
267 } else {
268 ScopedHashTableVal<K, V> *&KeyEntry = HT.TopLevelMap[ThisEntry->getKey()];
269 assert(KeyEntry == ThisEntry && "Scope imbalance!");
270 KeyEntry = ThisEntry->getNextForKey();
271 }
272
273 // Pop this value out of the scope.
274 LastValInScope = ThisEntry->getNextInScope();
275
276 // Delete this entry.
277 ThisEntry->Destroy(HT.getAllocator());
278 }
279}

281/// This method undoes the latest binding of the given key, effectively
282/// reverting to the previous state for that key. In the example at the
283/// beginning of this file, if we execute `HT.erase(0)` immediately after
284/// `HT.insert(0, 42);`, then the value associated with key "0" reverts to 0.
285/// This value is owned by "Scope1(HT)".
286template <typename K, typename V, typename KInfo, typename Allocator>
287void ScopedHashTableScope<K, V, KInfo, Allocator>::erase(const K &Key) {
288 auto It = HT.TopLevelMap.find(Key);
289 if (It == HT.TopLevelMap.end())
290 return;
291 ScopedHashTableVal<K, V> *&ThisEntry = It->second;
292
293 // `ThisEntry` may be the LastValInScope of a parent scope rather than the
294 // current scope. We iterate through the scope chain to find the scope
295 // that owns ThisEntry as its LastValInScope and update it accordingly.
296 auto *S = this;
297 while (S) {
298 if (ThisEntry == S->LastValInScope) {
299 S->LastValInScope = ThisEntry->getNextInScope();
300 break;
301 }
302 S = S->PrevScope;
303 }
304 if (ThisEntry->getNextForKey() == nullptr)
305 HT.TopLevelMap.erase(It);
306 ScopedHashTableVal<K, V>::erase(ThisEntry, HT.getAllocator());
307}
