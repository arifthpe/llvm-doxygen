77static MCSubtargetInfo *
78createAMDGPUMCSubtargetInfo(const Triple &TT, StringRef CPU, StringRef FS) {
79 if (TT.getArch() == Triple::r600)
80 return createR600MCSubtargetInfoImpl(TT, CPU, /*TuneCPU*/ CPU, FS);
81
82 StringRef ResolvedCPU = CPU;
83 if (ResolvedCPU.empty())
84 ResolvedCPU = AMDGPU::getArchNameFromSubArch(TT.getSubArch());
85
86 MCSubtargetInfo *STI = createAMDGPUMCSubtargetInfoImpl(
87 TT, ResolvedCPU, /*TuneCPU*/ ResolvedCPU, FS);
88
89 bool IsWave64 = STI->hasFeature(AMDGPU::FeatureWavefrontSize64);
90 bool IsWave32 = STI->hasFeature(AMDGPU::FeatureWavefrontSize32);
91
92 // FIXME: We should error for the default target.
93 if (STI->getFeatureBits().none())
94 STI->ToggleFeature(AMDGPU::FeatureSouthernIslands);
95
96 if (!IsWave64 && !IsWave32) {
97 // If there is no default wave size it must be a generation before gfx10,
98 // these have FeatureWavefrontSize64 in their definition already. For gfx10+
99 // set wave32 as a default.
100 STI->ToggleFeature(AMDGPU::isGFX10Plus(*STI)
101 ? AMDGPU::FeatureWavefrontSize32
102 : AMDGPU::FeatureWavefrontSize64);
103 } else if (IsWave64 && IsWave32) {
104 // The wave size is mutually exclusive. If both somehow end up set, wave32
105 // wins if supported.
106 STI->ToggleFeature(AMDGPU::supportsWave32(*STI)
107 ? AMDGPU::FeatureWavefrontSize64
108 : AMDGPU::FeatureWavefrontSize32);
109
110 // If both wavesizes were manually requested, hack in a feature to permit
111 // assembling modules with mixed wavesizes.
112 STI->ToggleFeature(AMDGPU::FeatureAssemblerPermissiveWavesize);
113 }
114
115 assert((STI->hasFeature(AMDGPU::FeatureWavefrontSize64) !=
116 STI->hasFeature(AMDGPU::FeatureWavefrontSize32)) &&
117 "wavesize features are mutually exclusive");
118
119 return STI;
120}

159bool AMDGPUMCInstrAnalysis::evaluateBranch(const MCInst &Inst, uint64_t Addr,
160 uint64_t Size,
161 uint64_t &Target) const {
162 if (Inst.getNumOperands() == 0 || !Inst.getOperand(0).isImm() ||
163 Info->get(Inst.getOpcode()).operands()[0].OperandType !=
164 MCOI::OPERAND_PCREL)
165 return false;
166
167 int64_t Imm = Inst.getOperand(0).getImm();
168 // Our branches take a simm16.
169 Target = SignExtend64<16>(Imm) * 4 + Addr + Size;
170 return true;
171}

173void AMDGPUMCInstrAnalysis::updateState(const MCInst &Inst,
174 const MCSubtargetInfo *STI,
175 uint64_t Addr) {
176 if (Inst.getOpcode() == AMDGPU::S_SET_VGPR_MSB_gfx12)
177 VgprMSBs = Inst.getOperand(0).getImm() & 0xff;
178 else if (Inst.getOpcode() == AMDGPU::S_SETREG_IMM32_B32_gfx12 &&
179 STI->hasFeature(AMDGPU::Feature1024AddressableVGPRs)) {
180 VgprMSBs = AMDGPU::convertSetRegImmToVgprMSBs(
181 Inst, STI->hasFeature(AMDGPU::FeatureSetregVGPRMSBFixup))
182 .value_or(0);
183 } else if (isTerminator(Inst))
184 VgprMSBs = 0;
185}

194extern "C" LLVM_ABI LLVM_EXTERNAL_VISIBILITY void
195LLVMInitializeAMDGPUTargetMC() {
196
197 TargetRegistry::RegisterMCInstrInfo(getTheGCNTarget(), createAMDGPUMCInstrInfo);
198 TargetRegistry::RegisterMCInstrInfo(getTheGCNLegacyTarget(),
199 createAMDGPUMCInstrInfo);
200 TargetRegistry::RegisterMCInstrInfo(getTheR600Target(),
201 createR600MCInstrInfo);
202 for (Target *T :
203 {&getTheR600Target(), &getTheGCNTarget(), &getTheGCNLegacyTarget()}) {
204 RegisterMCAsmInfo<AMDGPUMCAsmInfo> X(*T);
205
206 TargetRegistry::RegisterMCRegInfo(*T, createAMDGPUMCRegisterInfo);
207 TargetRegistry::RegisterMCSubtargetInfo(*T, createAMDGPUMCSubtargetInfo);
208 TargetRegistry::RegisterMCInstPrinter(*T, createAMDGPUMCInstPrinter);
209 TargetRegistry::RegisterMCInstrAnalysis(*T, createAMDGPUMCInstrAnalysis);
210 TargetRegistry::RegisterMCAsmBackend(*T, createAMDGPUAsmBackend);
211 TargetRegistry::RegisterELFStreamer(*T, createMCStreamer);
212 }
213
214 // R600 specific registration
215 TargetRegistry::RegisterMCCodeEmitter(getTheR600Target(),
216 createR600MCCodeEmitter);
217 TargetRegistry::RegisterObjectTargetStreamer(
218 getTheR600Target(), createAMDGPUObjectTargetStreamer);
219
220 // GCN specific registration
221 for (Target *T : {&getTheGCNTarget(), &getTheGCNLegacyTarget()}) {
222 TargetRegistry::RegisterMCCodeEmitter(*T, createAMDGPUMCCodeEmitter);
223
224 TargetRegistry::RegisterAsmTargetStreamer(*T,
225 createAMDGPUAsmTargetStreamer);
226 TargetRegistry::RegisterObjectTargetStreamer(
227 *T, createAMDGPUObjectTargetStreamer);
228 TargetRegistry::RegisterNullTargetStreamer(*T,
229 createAMDGPUNullTargetStreamer);
230 }
