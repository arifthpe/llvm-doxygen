22// This structure represents the information for a basic block pertaining to

29// The prefetch symbol is emitted immediately after the call of the given index,
30// in block `BBID` (First call has an index of 1). Zero callsite index means the
31// start of the block.
32struct CallsiteID {
33 UniqueBBID BBID;
34 unsigned CallsiteIndex;
35};

45// Provides DenseMapInfo for UniqueBBID.
46template <> struct DenseMapInfo<UniqueBBID> {
47 static unsigned getHashValue(const UniqueBBID &Val) {
48 return DenseMapInfo<unsigned>::getHashValue(Val.BaseID) ^
49 DenseMapInfo<unsigned>::getHashValue(Val.CloneID);
50 }
51
52 static bool isEqual(const UniqueBBID &LHS, const UniqueBBID &RHS) {
53 return DenseMapInfo<unsigned>::isEqual(LHS.BaseID, RHS.BaseID) &&
54 DenseMapInfo<unsigned>::isEqual(LHS.CloneID, RHS.CloneID);
