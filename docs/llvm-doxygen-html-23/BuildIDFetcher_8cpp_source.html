23DebuginfodFetcher::fetch(ArrayRef<uint8_t> BuildID) const {
24 Expected<std::string> Path = BuildIDFetcher::fetch(BuildID);
25 if (Path)
26 return Path;
27 // Not available locally, so try and download it now.
