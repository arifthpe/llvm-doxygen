30bool TargetSubtargetInfo::isIntrinsicSupported(unsigned IntrinsicID) const {
31 StringRef RequiredFeatures = Intrinsic::getRequiredTargetFeatures(
32 static_cast<Intrinsic::ID>(IntrinsicID));
33
34 if (RequiredFeatures.empty())
35 return true;
36
37 auto [It, Inserted] = IntrinsicSupportCache.try_emplace(IntrinsicID);
38 if (Inserted)
39 It->second = checkFeatureExpression(RequiredFeatures);
40 return It->second;
41}
