34 uint8_t AtLeast[65] = {};
35 uint64_t Boundaries[20] = {};
36 static constexpr int NumDigitsConstexpr(uint64_t N) {
37 int res = 1;
38 while (N >= 10) {
39 res++;
40 N /= 10;
41 }
42 return res;
43 }
44 constexpr ConstexprData() {

46 for (uint64_t i = 0; i <= 64; i++) {
47 uint64_t Digits = NumDigitsConstexpr(Val) - 1;
48 AtLeast[i] = Digits;

53 Val = 10;
