84struct Header {
85 uint8_t Magic[4]; // "DXBC"
86 Hash FileHash;
87 ContainerVersion Version;
88 uint32_t FileSize;
89 uint32_t PartCount;
90
91 void swapBytes() {
92 Version.swapBytes();
93 sys::swapByteOrder(FileSize);
94 sys::swapByteOrder(PartCount);
95 }
96 // Structure is followed by part offsets: uint32_t PartOffset[PartCount];
97 // The offset is to a PartHeader, which is followed by the Part Data.
98};

112struct BitcodeHeader {
113 uint8_t Magic[4]; // ACSII "DXIL".
114 uint8_t MinorVersion; // DXIL version.
115 uint8_t MajorVersion; // DXIL version.
116 uint16_t Unused;
117 uint32_t Offset; // Offset to LLVM bitcode (from start of header).
118 uint32_t Size; // Size of LLVM bitcode (in bytes).
119 // Followed by uint8_t[BitcodeHeader.Size] at &BitcodeHeader + Header.Offset
120
121 void swapBytes() {
122 sys::swapByteOrder(MinorVersion);
123 sys::swapByteOrder(MajorVersion);
124 sys::swapByteOrder(Offset);
125 sys::swapByteOrder(Size);
126 }
127};

129struct ProgramHeader {
130 uint8_t Version;
131 uint8_t Unused;
132 uint16_t ShaderKind;
133 uint32_t Size; // Size in uint32_t words including this header.
134 BitcodeHeader Bitcode;
135
136 void swapBytes() {
137 sys::swapByteOrder(ShaderKind);
138 sys::swapByteOrder(Size);
139 Bitcode.swapBytes();
140 }
141 uint8_t getMajorVersion() { return Version >> 4; }
142 uint8_t getMinorVersion() { return Version & 0xF; }
143 static uint8_t getVersion(uint8_t Major, uint8_t Minor) {
144 return (Major << 4) | Minor;
145 }
146};

349union PipelinePSVInfo {
350 VertexPSVInfo VS;

354 PixelPSVInfo PS;
355 MeshPSVInfo MS;

358 void swapBytes(Triple::EnvironmentType Stage) {
359 switch (Stage) {
360 case Triple::EnvironmentType::Pixel:
361 PS.swapBytes();
362 break;
363 case Triple::EnvironmentType::Vertex:
364 VS.swapBytes();
365 break;
366 case Triple::EnvironmentType::Geometry:
367 GS.swapBytes();
368 break;
369 case Triple::EnvironmentType::Hull:
370 HS.swapBytes();
371 break;
372 case Triple::EnvironmentType::Domain:
373 DS.swapBytes();
374 break;
375 case Triple::EnvironmentType::Mesh:
376 MS.swapBytes();
377 break;
378 case Triple::EnvironmentType::Amplification:
379 AS.swapBytes();
380 break;
381 default:
382 break;
383 }
384 }
385};

620struct ProgramSignatureElement {
621 uint32_t Stream; // Stream index (parameters must appear in non-decreasing
622 // stream order)
623 uint32_t NameOffset; // Offset from the start of the ProgramSignatureHeader to
624 // the start of the null terminated string for the name.
625 uint32_t Index; // Semantic Index
626 D3DSystemValue SystemValue; // Semantic type. Similar to PSV::SemanticKind.
627 SigComponentType CompType; // Type of bits.
628 uint32_t Register; // Register Index (row index)
629 uint8_t Mask; // Mask (column allocation)
630
631 // The ExclusiveMask has a different meaning for input and output signatures.
632 // For an output signature, masked components of the output register are never
633 // written to.
634 // For an input signature, masked components of the input register are always
635 // read.
636 uint8_t ExclusiveMask;
637
638 uint16_t Unused;
639 SigMinPrecision MinPrecision; // Minimum precision of input/output data
640
641 void swapBytes() {
642 sys::swapByteOrder(Stream);
643 sys::swapByteOrder(NameOffset);
644 sys::swapByteOrder(Index);
645 sys::swapByteOrder(SystemValue);
646 sys::swapByteOrder(CompType);
647 sys::swapByteOrder(Register);
648 sys::swapByteOrder(Mask);
649 sys::swapByteOrder(ExclusiveMask);
650 sys::swapByteOrder(MinPrecision);
651 }
652};
