718 MachineRegisterInfo &MRI,
719 LegalizerHelper &Helper) const {
720 MachineIRBuilder &MIRBuilder = Helper.MIRBuilder;
721 const auto &BuildVector = cast<GBuildVector>(MI);
722 Register Dst = BuildVector.getReg(0);
723 LLT DstTy = MRI.getType(Dst);
724 MachineFunction &MF = MIRBuilder.getMF();
725 LLVMContext &Ctx = MF.getFunction().getContext();
726 uint64_t DstTySize = DstTy.getScalarSizeInBits();
727
728 SmallVector<Constant *, 4> CstIdxs;
729 for (unsigned i = 0; i < BuildVector.getNumSources(); ++i) {
730 Register Source = BuildVector.getSourceReg(i);

732 auto ValueAndReg = getIConstantVRegValWithLookThrough(Source, MRI);
733 if (ValueAndReg) {
734 CstIdxs.emplace_back(ConstantInt::get(Ctx, ValueAndReg->Value));
735 continue;

737
738 auto FPValueAndReg = getFConstantVRegValWithLookThrough(Source, MRI);
739 if (FPValueAndReg) {
740 CstIdxs.emplace_back(ConstantFP::get(Ctx, FPValueAndReg->Value));
741 continue;
742 }
743
744 if (getOpcodeDef<GImplicitDef>(Source, MRI)) {
745 CstIdxs.emplace_back(UndefValue::get(Type::getIntNTy(Ctx, DstTySize)));

750
751 Constant *ConstVal = ConstantVector::get(CstIdxs);
752
753 const DataLayout &DL = MIRBuilder.getDataLayout();

760 MF.getMachineMemOperand(MachinePointerInfo::getConstantPool(MF),
761 MachineMemOperand::MOLoad, DstTy, Alignment);

763 MIRBuilder.buildLoad(Dst, Addr, *MMO);
764 MI.eraseFromParent();
765 return true;

823bool X86LegalizerInfo::legalizeGETROUNDING(MachineInstr &MI,
824 MachineRegisterInfo &MRI,
825 LegalizerHelper &Helper) const {
826 /*
827 The rounding mode is in bits 11:10 of FPSR, and has the following
828 settings:
829 00 Round to nearest
830 01 Round to -inf
831 10 Round to +inf
832 11 Round to 0
833
834 GET_ROUNDING, on the other hand, expects the following:
835 -1 Undefined
836 0 Round to 0
837 1 Round to nearest
838 2 Round to +inf
839 3 Round to -inf
840
841 To perform the conversion, we use a packed lookup table of the four 2-bit
842 values that we can index by FPSP[11:10]
843 0x2d --> (0b00,10,11,01) --> (0,2,3,1) >> FPSR[11:10]
844
845 (0x2d >> ((FPSR >> 9) & 6)) & 3
846 */
847
848 MachineIRBuilder &MIRBuilder = Helper.MIRBuilder;
849 MachineFunction &MF = MIRBuilder.getMF();
850 Register Dst = MI.getOperand(0).getReg();
851 LLT DstTy = MRI.getType(Dst);
852 const LLT s8 = LLT::scalar(8);
853 const LLT s16 = LLT::scalar(16);
854 const LLT s32 = LLT::scalar(32);
855
856 // Save FP Control Word to stack slot
857 int MemSize = 2;
858 Align Alignment = Align(2);
859 MachinePointerInfo PtrInfo;
860 auto StackTemp = Helper.createStackTemporary(TypeSize::getFixed(MemSize),
861 Alignment, PtrInfo);
862 Register StackPtr = StackTemp.getReg(0);
863
864 auto StoreMMO = MF.getMachineMemOperand(PtrInfo, MachineMemOperand::MOStore,
865 MemSize, Alignment);
866
867 // Store FP Control Word to stack slot using G_FNSTCW16
868 MIRBuilder.buildInstr(X86::G_FNSTCW16)
869 .addUse(StackPtr)
870 .addMemOperand(StoreMMO);
871
872 // Load FP Control Word from stack slot
873 auto LoadMMO = MF.getMachineMemOperand(PtrInfo, MachineMemOperand::MOLoad,
874 MemSize, Alignment);
875
876 auto CWD32 =
877 MIRBuilder.buildZExt(s32, MIRBuilder.buildLoad(s16, StackPtr, *LoadMMO));
878 auto Shifted8 = MIRBuilder.buildTrunc(
879 s8, MIRBuilder.buildLShr(s32, CWD32, MIRBuilder.buildConstant(s8, 9)));
880 auto Masked32 = MIRBuilder.buildZExt(
881 s32, MIRBuilder.buildAnd(s8, Shifted8, MIRBuilder.buildConstant(s8, 6)));
882
883 // LUT is a packed lookup table (0x2d) used to map the 2-bit x87 FPU rounding
884 // mode (from bits 11:10 of the control word) to the values expected by
885 // GET_ROUNDING. The mapping is performed by shifting LUT right by the
886 // extracted rounding mode and masking the result with 3 to obtain the final
887 auto LUT = MIRBuilder.buildConstant(s32, 0x2d);
888 auto LUTShifted = MIRBuilder.buildLShr(s32, LUT, Masked32);
889 auto RetVal =
890 MIRBuilder.buildAnd(s32, LUTShifted, MIRBuilder.buildConstant(s32, 3));
891 auto RetValTrunc = MIRBuilder.buildZExtOrTrunc(DstTy, RetVal);
892
893 MIRBuilder.buildCopy(Dst, RetValTrunc);
894
895 MI.eraseFromParent();
896 return true;
897}

899bool X86LegalizerInfo::legalizeSETROUNDING(MachineInstr &MI,
900 MachineRegisterInfo &MRI,
901 LegalizerHelper &Helper) const {
902 MachineIRBuilder &MIRBuilder = Helper.MIRBuilder;
903 MachineFunction &MF = MIRBuilder.getMF();
904 Register Src = MI.getOperand(0).getReg();
905 const LLT s8 = LLT::scalar(8);
906 const LLT s16 = LLT::scalar(16);
907 const LLT s32 = LLT::scalar(32);
908
909 // Allocate stack slot for control word and MXCSR (4 bytes).
910 int MemSize = 4;
911 Align Alignment = Align(4);
912 MachinePointerInfo PtrInfo;
913 auto StackTemp = Helper.createStackTemporary(TypeSize::getFixed(MemSize),
914 Alignment, PtrInfo);
915 Register StackPtr = StackTemp.getReg(0);
916
917 auto StoreMMO =
918 MF.getMachineMemOperand(PtrInfo, MachineMemOperand::MOStore, 2, Align(2));
919 MIRBuilder.buildInstr(X86::G_FNSTCW16)
920 .addUse(StackPtr)
921 .addMemOperand(StoreMMO);
922
923 auto LoadMMO =
924 MF.getMachineMemOperand(PtrInfo, MachineMemOperand::MOLoad, 2, Align(2));
925 auto CWD16 = MIRBuilder.buildLoad(s16, StackPtr, *LoadMMO);
926
927 // Clear RM field (bits 11:10)
928 auto ClearedCWD =
929 MIRBuilder.buildAnd(s16, CWD16, MIRBuilder.buildConstant(s16, 0xf3ff));
930
931 // Check if Src is a constant
932 auto *SrcDef = MRI.getVRegDef(Src);
933 Register RMBits;
934 Register MXCSRRMBits;
935
936 if (SrcDef && SrcDef->getOpcode() == TargetOpcode::G_CONSTANT) {
937 uint64_t RM = getIConstantFromReg(Src, MRI).getZExtValue();
938 int FieldVal = X86::getRoundingModeX86(RM);
939
940 if (FieldVal == X86::rmInvalid) {
941 FieldVal = X86::rmToNearest;
942 LLVMContext &C = MF.getFunction().getContext();
943 C.diagnose(DiagnosticInfoUnsupported(
944 MF.getFunction(), "rounding mode is not supported by X86 hardware",
945 DiagnosticLocation(MI.getDebugLoc()), DS_Error));
946 return false;
947 }
948
949 FieldVal = FieldVal << 3;
950 RMBits = MIRBuilder.buildConstant(s16, FieldVal).getReg(0);
951 MXCSRRMBits = MIRBuilder.buildConstant(s32, FieldVal).getReg(0);
952 } else {
953 // Convert Src (rounding mode) to bits for control word
954 // (0xc9 << (2 * Src + 4)) & 0xc00
955 auto Src32 = MIRBuilder.buildZExtOrTrunc(s32, Src);
956 auto ShiftAmt = MIRBuilder.buildAdd(
957 s32, MIRBuilder.buildShl(s32, Src32, MIRBuilder.buildConstant(s32, 1)),
958 MIRBuilder.buildConstant(s32, 4));
959 auto ShiftAmt8 = MIRBuilder.buildTrunc(s8, ShiftAmt);
960 auto Shifted = MIRBuilder.buildShl(s16, MIRBuilder.buildConstant(s16, 0xc9),
961 ShiftAmt8);
962 RMBits =
963 MIRBuilder.buildAnd(s16, Shifted, MIRBuilder.buildConstant(s16, 0xc00))
964 .getReg(0);
965
966 // For non-constant case, we still need to compute MXCSR bits dynamically
967 auto RMBits32 = MIRBuilder.buildZExt(s32, RMBits);
968 MXCSRRMBits =
969 MIRBuilder.buildShl(s32, RMBits32, MIRBuilder.buildConstant(s32, 3))
970 .getReg(0);
971 }
972 // Update rounding mode bits
973 auto NewCWD =
974 MIRBuilder.buildOr(s16, ClearedCWD, RMBits, MachineInstr::Disjoint);
975
976 // Store new FP Control Word to stack
977 auto StoreNewMMO =
978 MF.getMachineMemOperand(PtrInfo, MachineMemOperand::MOStore, 2, Align(2));
979 MIRBuilder.buildStore(NewCWD, StackPtr, *StoreNewMMO);
980
981 // Load FP control word from the slot using G_FLDCW16
982 auto LoadNewMMO =
983 MF.getMachineMemOperand(PtrInfo, MachineMemOperand::MOLoad, 2, Align(2));
984 MIRBuilder.buildInstr(X86::G_FLDCW16)
985 .addUse(StackPtr)
986 .addMemOperand(LoadNewMMO);
987
988 if (Subtarget.hasSSE1()) {
989 // Store MXCSR to stack (use STMXCSR)
990 auto StoreMXCSRMMO = MF.getMachineMemOperand(
991 PtrInfo, MachineMemOperand::MOStore, 4, Align(4));
992 MIRBuilder.buildInstr(TargetOpcode::G_INTRINSIC_W_SIDE_EFFECTS)
993 .addIntrinsicID(Intrinsic::x86_sse_stmxcsr)
994 .addUse(StackPtr)
995 .addMemOperand(StoreMXCSRMMO);
996
997 // Load MXCSR from stack
998 auto LoadMXCSRMMO = MF.getMachineMemOperand(
999 PtrInfo, MachineMemOperand::MOLoad, 4, Align(4));
1000 auto MXCSR = MIRBuilder.buildLoad(s32, StackPtr, *LoadMXCSRMMO);
1001
1002 // Clear RM field (bits 14:13)
1003 auto ClearedMXCSR = MIRBuilder.buildAnd(
1004 s32, MXCSR, MIRBuilder.buildConstant(s32, 0xffff9fff));
1005
1006 // Update rounding mode bits
1007 auto NewMXCSR = MIRBuilder.buildOr(s32, ClearedMXCSR, MXCSRRMBits);
1008
1009 // Store new MXCSR to stack
1010 auto StoreNewMXCSRMMO = MF.getMachineMemOperand(
1011 PtrInfo, MachineMemOperand::MOStore, 4, Align(4));
1012 MIRBuilder.buildStore(NewMXCSR, StackPtr, *StoreNewMXCSRMMO);
1013
1014 // Load MXCSR from stack (use LDMXCSR)
1015 auto LoadNewMXCSRMMO = MF.getMachineMemOperand(
1016 PtrInfo, MachineMemOperand::MOLoad, 4, Align(4));
1017 MIRBuilder.buildInstr(TargetOpcode::G_INTRINSIC_W_SIDE_EFFECTS)
1018 .addIntrinsicID(Intrinsic::x86_sse_ldmxcsr)
1019 .addUse(StackPtr)
1020 .addMemOperand(LoadNewMXCSRMMO);
1021 }
1022
1023 MI.eraseFromParent();
1024 return true;
1025}
