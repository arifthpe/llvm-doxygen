69uint8_t WebAssemblyMCCodeEmitter::getEncodedMemOrder(uint8_t Order,
70 unsigned Opcode) const {
71 if (Order == wasm::WASM_MEM_ORDER_ACQ_REL) {
72 StringRef Name = MCII.getName(Opcode);
73 if (Name.contains("RMW") || Name.contains("CMPXCHG"))
74 return wasm::WASM_MEM_ORDER_RMW_ACQ_REL;
75 }
76 return Order;
77}

79void WebAssemblyMCCodeEmitter::encodeP2AlignAndMemOrder(
80 const MCInst &MI, unsigned P2AlignIdx, const MCInstrDesc &Desc,
81 const MCSubtargetInfo &STI, raw_ostream &OS,
82 SmallVectorImpl<MCFixup> &Fixups, uint64_t Start) const {
83 uint64_t P2Align = MI.getOperand(P2AlignIdx).getImm();
84 uint8_t Order = wasm::WASM_MEM_ORDER_SEQ_CST;
85
86 // Atomic instructions always have an ordering, but if it's SEQ_CST then we
87 // don't use the relaxed-atomics encoding (even if relaxed-atomics is
88 // enabled) because the original encoding is smaller.
89 if (P2AlignIdx > 0 && Desc.operands()[P2AlignIdx - 1].OperandType ==
90 WebAssembly::OPERAND_MEMORDER) {
91 Order = MI.getOperand(P2AlignIdx - 1).getImm();
92 if (Order != wasm::WASM_MEM_ORDER_SEQ_CST) {
93 assert(STI.getFeatureBits()[WebAssembly::FeatureRelaxedAtomics] &&
94 "Non-default atomic ordering but feature not enabled");
95 P2Align |= wasm::WASM_MEMARG_HAS_MEM_ORDER;
96 }
97 }
98
99 encodeULEB128(P2Align, OS);
100
101 // Memory index will go here once we support multi-memory.
102
103 if (P2Align & wasm::WASM_MEMARG_HAS_MEM_ORDER) {
104 support::endian::write<uint8_t>(OS,
105 getEncodedMemOrder(Order, MI.getOpcode()),
106 llvm::endianness::little);
107 }
108}

110void WebAssemblyMCCodeEmitter::encodeInstruction(
111 const MCInst &MI, SmallVectorImpl<char> &CB,
112 SmallVectorImpl<MCFixup> &Fixups, const MCSubtargetInfo &STI) const {
113 raw_svector_ostream OS(CB);
114 uint64_t Start = OS.tell();
115
116 uint64_t Binary = getBinaryCodeForInstr(MI, Fixups, STI);
117 if (Binary < (1 << 8)) {
118 OS << uint8_t(Binary);
119 } else if (Binary < (1 << 16)) {
120 OS << uint8_t(Binary >> 8);
121 encodeULEB128(uint8_t(Binary), OS);
122 } else if (Binary < (1 << 24)) {
123 OS << uint8_t(Binary >> 16);
124 encodeULEB128(uint16_t(Binary), OS);
125 } else {
126 llvm_unreachable("Very large (prefix + 3 byte) opcodes not supported");
127 }
128
129 // For br_table instructions, encode the size of the table. In the MCInst,
130 // there's an index operand (if not a stack instruction), one operand for
131 // each table entry, and the default operand.
132 unsigned Opcode = MI.getOpcode();
133 if (Opcode == WebAssembly::BR_TABLE_I32_S ||
134 Opcode == WebAssembly::BR_TABLE_I64_S)
135 encodeULEB128(MI.getNumOperands() - 1, OS);
136 if (Opcode == WebAssembly::BR_TABLE_I32 ||
137 Opcode == WebAssembly::BR_TABLE_I64)
138 encodeULEB128(MI.getNumOperands() - 2, OS);
139
140 const MCInstrDesc &Desc = MCII.get(Opcode);
141 for (unsigned I = 0, E = MI.getNumOperands(); I < E; ++I) {
142 const MCOperand &MO = MI.getOperand(I);
143 if (MO.isReg()) {
144 /* nothing to encode */
145
146 } else if (MO.isImm()) {
147 if (I < Desc.getNumOperands()) {
148 const MCOperandInfo &Info = Desc.operands()[I];
149 LLVM_DEBUG(dbgs() << "Encoding immediate: type="
150 << int(Info.OperandType) << "\n");
151 switch (Info.OperandType) {
152 case WebAssembly::OPERAND_I32IMM:
153 encodeSLEB128(int32_t(MO.getImm()), OS);
154 break;
155 case WebAssembly::OPERAND_P2ALIGN:
156 encodeP2AlignAndMemOrder(MI, I, Desc, STI, OS, Fixups, Start);
157 break;
158 case WebAssembly::OPERAND_OFFSET32:
159 encodeULEB128(uint32_t(MO.getImm()), OS);
160 break;
161 case WebAssembly::OPERAND_I64IMM:
162 encodeSLEB128(MO.getImm(), OS);
163 break;
164 case WebAssembly::OPERAND_SIGNATURE:
165 case WebAssembly::OPERAND_VEC_I8IMM:
166 support::endian::write<uint8_t>(OS, MO.getImm(),
167 llvm::endianness::little);
168 break;
169 case WebAssembly::OPERAND_VALTYPE_LIST: {
170 // A vec of valtypes: emit the ULEB count followed by that many
171 // single-byte value types. The count lives in this MCOperand;
172 // subsequent MCOperands hold the valtype bytes and are consumed here.
173 uint64_t Count = uint64_t(MO.getImm());
174 if (Count > uint64_t(E - I - 1))
175 report_fatal_error(
176 "select t* operand count exceeds MCInst operand count");
177 encodeULEB128(Count, OS);
178 for (uint64_t K = 0; K < Count; ++K) {
179 ++I;
180 int64_t Val = MI.getOperand(I).getImm();
181 assert(isUInt<8>(Val) && "select t* valtype out of byte range");
182 support::endian::write<uint8_t>(OS, Val, llvm::endianness::little);
183 }
184 break;
185 }
186 case WebAssembly::OPERAND_MEMORDER: {
187 // If there is a p2align operand (everything but fence) it is encoded
188 // together with the mem ordering (in the next iteration).
189 if (I + 1 < Desc.getNumOperands() &&
190 Desc.operands()[I + 1].OperandType ==
191 WebAssembly::OPERAND_P2ALIGN)
192 break;
193 uint8_t Val = getEncodedMemOrder(MO.getImm(), Opcode);
194 if (STI.getFeatureBits()[WebAssembly::FeatureRelaxedAtomics]) {
195 support::endian::write<uint8_t>(OS, Val, llvm::endianness::little);
196 } else {
197 assert(Opcode == WebAssembly::ATOMIC_FENCE_S);
198 support::endian::write<uint8_t>(OS, 0, llvm::endianness::little);
199 }
200 break;
201 }
202 case WebAssembly::OPERAND_VEC_I16IMM:
203 support::endian::write<uint16_t>(OS, MO.getImm(),
204 llvm::endianness::little);
205 break;
206 case WebAssembly::OPERAND_VEC_I32IMM:
207 support::endian::write<uint32_t>(OS, MO.getImm(),
208 llvm::endianness::little);
209 break;
210 case WebAssembly::OPERAND_VEC_I64IMM:
211 support::endian::write<uint64_t>(OS, MO.getImm(),
212 llvm::endianness::little);
213 break;
214 case WebAssembly::OPERAND_GLOBAL:
215 Ctx.reportError(
216 SMLoc(),
217 Twine("Wasm globals should only be accessed symbolically!"));
218 break;
219 default:
220 encodeULEB128(uint64_t(MO.getImm()), OS);
221 }
222 } else {
223 // Variadic immediate operands are br_table's destination operands or
224 // try_table's operands (# of catch clauses, catch sub-opcodes, or catch
225 // clause destinations)
226 assert(WebAssembly::isBrTable(Opcode) ||
227 Opcode == WebAssembly::TRY_TABLE_S);
228 encodeULEB128(uint32_t(MO.getImm()), OS);
229 }
230
231 } else if (MO.isSFPImm()) {
232 uint32_t F = MO.getSFPImm();
233 support::endian::write<uint32_t>(OS, F, llvm::endianness::little);
234 } else if (MO.isDFPImm()) {
235 uint64_t D = MO.getDFPImm();
236 support::endian::write<uint64_t>(OS, D, llvm::endianness::little);
237 } else if (MO.isExpr()) {
238 llvm::MCFixupKind FixupKind;
239 size_t PaddedSize = 5;
240 if (I < Desc.getNumOperands()) {
241 const MCOperandInfo &Info = Desc.operands()[I];
242 switch (Info.OperandType) {
243 case WebAssembly::OPERAND_I32IMM:
244 FixupKind = WebAssembly::fixup_sleb128_i32;
245 break;
246 case WebAssembly::OPERAND_I64IMM:
247 FixupKind = WebAssembly::fixup_sleb128_i64;
248 PaddedSize = 10;
249 break;
250 case WebAssembly::OPERAND_FUNCTION32:
251 case WebAssembly::OPERAND_TABLE:
252 case WebAssembly::OPERAND_OFFSET32:
253 case WebAssembly::OPERAND_SIGNATURE:
254 case WebAssembly::OPERAND_TYPEINDEX:
255 case WebAssembly::OPERAND_GLOBAL:
256 case WebAssembly::OPERAND_TAG:
257 FixupKind = WebAssembly::fixup_uleb128_i32;
258 break;
259 case WebAssembly::OPERAND_OFFSET64:
260 FixupKind = WebAssembly::fixup_uleb128_i64;
261 PaddedSize = 10;
262 break;
263 default:
264 llvm_unreachable("unexpected symbolic operand kind");
265 }
266 } else {
267 // Variadic expr operands are try_table's catch/catch_ref clauses' tags.
268 assert(Opcode == WebAssembly::TRY_TABLE_S);
269 FixupKind = WebAssembly::fixup_uleb128_i32;
270 }
271 Fixups.push_back(
272 MCFixup::create(OS.tell() - Start, MO.getExpr(), FixupKind));
273 ++MCNumFixups;
274 encodeULEB128(0, OS, PaddedSize);
275 } else {
276 llvm_unreachable("unexpected operand kind");
277 }
278 }
279
280 ++MCNumEmitted; // Keep track of the # of mi's emitted.
281}
